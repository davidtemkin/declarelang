/*
# The Venue

*The Marriage of Figaro*, twenty performances in the Grand Hall: pick one,
choose seats in the hall, pay, and keep the tickets.

How it is put together:

- **The records are the service's**, each a `DataSource` on the App: the
  production, the seat map of the performance being looked at (kept current
  by `feed`, the stream of other people's purchases), the account, and the
  three writes — signing in, signing out, paying. Every request carries the
  session cookie (`credentials = include`).
- **Where the person is** is a handful of attributes: the screen, the
  performance, the seats chosen (`chosen`, ids), the section in view on a
  phone, whether the sign-in sheet is up.
- **What the checkout form edits** is `pay`, a `Payment` (venue-model), so a
  sign-in over the form, or a trip back to the hall, loses nothing.
- **Everything else derives**: the picks and their totals from `chosen` and
  the seat map; each seat's state from its record and `chosen`.

The hall is in `venue-hall.declare`, the parts of each screen in
`venue-parts.declare`, the data's shapes and the formatting helpers in
`venue-model.declare`.
*/

    tierOf(key: string) -> number { return Math.max(0, app.sections.findIndex((s) => s.key == key)) },
    firstRowOf(key: string) -> number {
        let n = 0
        for (const s of app.sections) { if (s.key == key) return n; n = n + s.rows.length }
        return n
        },
    freeIn(key: string) -> number {
        const s = app.sections.find((x) => x.key == key)
        if (s == null) return 0
        return s.rows.reduce((m, r) => m + r.seats.filter((x) => x.status == "available").length, 0)
        },
    mineIn(key: string) -> number { return app.picks.filter((p) => p.id.startsWith(key + "-")).length },
    freeSeats: number = { app.sections.reduce((m, s) => m + app.freeIn(s.key), 0) },
    allSeats: number = { app.sections.reduce((m, s) => m + s.rows.reduce((n, r) => n + r.seats.length, 0), 0) },
    sectionHeight(s: Section?, pitch: number) -> number {
        if (s == null) return 0
        const widest = s.rows.reduce((m, r) => Math.max(m, r.seats.length), 0)
        return sagAtTop(widest, app.firstRowOf(s.key), pitch) + s.rows.length * pitch
        },

    // One seat by its id: where it is and what it costs.

    pickOf(id: string) -> SeatPick? {
        for (const s of app.sections) {
            for (const r of s.rows) {
                const seat = r.seats.find((x) => x.id == id)
                if (seat != null) return ({ id: id, section: s.name, row: r.name, number: seat.number, priceCents: s.priceCents, feeCents: s.feeCents })
                }
            }
        return null
        },
    seatLabel(id: string) -> string {
        const p = app.pickOf(id)
        return p == null ? id : p.section + ", row " + p.row + ", seat " + p.number
        },

    byMonth() -> object {
        const out: { name: string, id: string, perfs: Performance[] }[] = []

            const name = monthName(p.startsAt)
            if (out.length == 0 || out[out.length - 1].name != name) out.push(({ name: name, id: name, perfs: [] }))
            out[out.length - 1].perfs.push(p)
            }
        return ({ months: out })
        },

    // ── the verbs

    openPerformance(id: string) {
        if (id != perfId) { chosen = []; phoneSection = 0 }
        perfId = id
        notice = ""
        screen = "hall"
        seatMap.fetch()
        scrollTo(0)
        },
    toggleSeat(id: string) {
        if (chosen.includes(id)) { chosen = chosen.filter((c) => c != id); return }
        const p = pickOf(id)
        if (p == null) return
        chosen = [...chosen, id]
        },
    clearSeats() { chosen = [] },

    // A seat someone else bought: it goes, and if it was this person's, they
    // are told which.

    seatGone(id: string) {
        const all = seatMap.value?.sections ?? []
        all.forEach((s, si) => s.rows.forEach((r, ri) => r.seats.forEach((x, k) => {
            if (x.id == id && x.status != "sold") seatMap.set(["sections", si, "rows", ri, "seats", k, "status"], "sold")
            })))
        if (chosen.includes(id)) {
            const label = seatLabel(id)
            chosen = chosen.filter((c) => c != id)
            tell(label + " has just been taken by someone else.")
            if (chosen.length == 0 && screen == "checkout") screen = "hall"
            }
        },

    // The payment found seats gone: nothing was charged; back to the hall
    // with the rest of the choice.

    seatsTaken(ids: string[], message: string) {
        chosen = chosen.filter((c) => !ids.includes(c))
        notice = message
        screen = "hall"
        seatMap.fetch()
        scrollTo(0)
        Accessibility.announce(message, "assertive")
        },
    tell(message: string) {
        notice = message
        toast.say(message)
        },

    toCheckout() {
        notice = ""
        screen = "checkout"
        scrollTo(0)
        afterSettle(() => Focus.focus(app.checkoutTitle))
        },
    backToHall() { screen = "hall"; scrollTo(0) },
    backToList() { screen = "list"; production.fetch(); scrollTo(0) },
    paid() {
        chosen = []
        notice = ""
        toast.close()
        pay.forgetCard()
        screen = "tickets"
        production.fetch()
        scrollTo(0)
        afterSettle(() => Focus.focus(app.ticketsTitle))
        },
    startOver() { perfId = ""; screen = "list"; order.clear(); production.fetch(); scrollTo(0) },
    signOut() { signOutRequest.fetch() },

    // The arrows walked off the end of a section: on into the next one.

    leaveSection(key: string, dy: number, offset: number) {
        const at = sections.findIndex((s) => s.key == key) + dy
        if (at < 0 || at >= sections.length) return
        if (phone) {
            phoneSection = at
            afterSettle(() => app.enterSection(app.phonePlan, dy, offset))
            return
            }
        enterSection((deskHouse?.childViews[at] ?? null) as SectionPlan | null, dy, offset)
        },
    enterSection(plan: SectionPlan?, dy: number, offset: number) {
        if (plan == null) return
        const n = plan.rowList.length
        let r = dy > 0 ? 0 : n - 1
        while (r >= 0 && r < n) {
            const k = plan.nearest(r, offset)
            if (k >= 0) { plan.focusSeat(r, k); return }
            r = r + dy
            }
        },
