class Checkout extends Form [ width = 100%, source = { app.order },
    layout: SimpleLayout [ axis = y, spacing = 32 ],
    failedMessage = "The payment couldn't be sent. Nothing was charged; try again.",
    uncertainMessage = "We didn't hear back about this payment. Check your email before trying again.",

    // A lost seat is answered in the hall, not on a field.

    refusal(body: Refusal) -> FormProblem[] {
        if (body.error == "unavailable") return []
        return super.refusal(body)
        },
    refusedWith(body: Refusal) {
        if (body.error == "unavailable") { app.seatsTaken(body.conflicts ?? [], body.message ?? ""); return }
        if (body.error == "card_declined" || body.error == "insufficient_funds") {
            // Everything stays but the security code, which reads as new.
            app.pay.cvc = ""
            cards.savedCodeRow.savedField.savedCode.judged = false
            card.codes.newCodeField.newCode.judged = false
            }
        if (body.error == "signed_out") app.me.fetch()
        },

    // ── how to pay: a saved card, or a new one

    cards: FormPart [ title = "Pay with", visible = { app.user != null },
        choice: RadioGroup [ width = 100%, accessibilityLabel = "Pay with",
            datapath = { app.cardList.value },
            value = { app.pay.method },
            commit(v: string) { app.pay.method = v },
            layout: SimpleLayout [ axis = y, spacing = 10 ],
            CardChoice [ datapath = :cards[], choice = :id, brand = :brand, last4 = :last4,
                expMonth = :expMonth, expYear = :expYear, expired = { app.cardExpired(:expMonth ?? 0, :expYear ?? 0) } ],
            CardChoice [ choice = "new", newCard = true ]
            ],
        savedCodeRow: View [ width = 100%, visible = { app.pay.useSaved },
            savedField: QuietField [ width = { app.phone ? parent.width : 220 },
                label = "Security code",
                savedCode: TextInput [ width = 100%, name = "cvc", purpose = cardCode,
                    required = "Enter the security code from the back of the card.",
                    value <-> app.pay.cvc,
                    admit(t: string) -> string | null { return /^\d*$/.test(t) && t.length <= codeLength(app.pay.brand) ? t : null },
                    validate(v: string) -> string { return v.length == codeLength(app.pay.brand) ? "" : "The security code is " + codeLength(app.pay.brand) + " digits." }
                    ]
                ]
            ]
        ],

    // ── a new card

    card: FormPart [ title = { app.user != null ? "New card" : "Payment" }, visible = { !app.pay.useSaved },
        WideField [ label = "Email",
            TextInput [ width = 100%, name = "email", purpose = email, required = "Enter the email address to send the tickets to.",
                value <-> app.pay.email,
                validate(v: string) -> string { return emailProblem(v) }
                ]
            ],
        Text [ width = 100%, fontSize = 13, textColor = { provided("theme").textMuted }, text = "Your tickets are sent here." ],
        WideField [ label = "Name on card",
            TextInput [ width = 100%, name = "name", purpose = cardName, required = "Enter the name as it's printed on the card.",
                value <-> app.pay.name ]
            ],
        WideField [ label = "Card number",
            cardNumber: TextInput [ width = { parent.width - 76 }, name = "number", purpose = cardNumber,
                required = "Enter the card number.",
                numeralWidth = tabular,
                value <-> app.pay.number,
                admit(t: string) -> string | null {
                    if (!/^[\d ]*$/.test(t)) return null
                    return groupCard(digitsOf(t))
                    },
                validate(v: string) -> string { return cardProblem(digitsOf(v)) }
                ],
            BrandMark [ brand = { cardBrand(digitsOf(app.pay.number)) } ]
            ],
        codes: Pair [
            StackedField [ label = "Expiry",
                TextInput [ width = 100%, name = "expiry", purpose = cardExpiry, required = "Enter the expiry date.",
                    placeholder = "MM / YY", numeralWidth = tabular,
                    value <-> app.pay.expiry,
                    admit(t: string) -> string | null { return /^[\d /]*$/.test(t) ? groupExpiry(t) : null },
                    validate(v: string) -> string { return expiryProblem(v, app.clock.year, app.clock.month) }
                    ]
                ],
            newCodeField: StackedField [ label = "Security code",
                newCode: TextInput [ width = 100%, name = "cvc", purpose = cardCode,
                    required = "Enter the security code.", numeralWidth = tabular,
                    placeholder = { codeLength(app.pay.brand) == 4 ? "4 digits" : "3 digits" },
                    value <-> app.pay.cvc,
                    admit(t: string) -> string | null { return /^\d*$/.test(t) && t.length <= 4 ? t : null },
                    validate(v: string) -> string {
                        const n = codeLength(app.pay.brand)
                        return v.length == n ? "" : n == 4 ? "American Express codes are 4 digits, on the front." : "The security code is 3 digits, on the back."
                        }
                    ]
                ]
            ]
        ],

    // ── where the card is billed

    billing: FormPart [ title = "Billing address", visible = { !app.pay.useSaved },
        WideField [ label = "Street address",
            TextInput [ width = 100%, name = "line1", purpose = addressLine1, required = "Enter the street address.",
                value <-> app.pay.line1 ]
            ],
        WideField [ label = "Apartment, suite (optional)",
            TextInput [ width = 100%, name = "line2", purpose = addressLine2, value <-> app.pay.line2 ]
            ],
        WideField [ label = "City",
            TextInput [ width = 100%, name = "city", purpose = city, required = "Enter the city.", value <-> app.pay.city ]
            ],
        Pair [
            StackedField [ label = "Country",
                Combobox [ width = 100%, name = "country", purpose = country, required = "Choose a country.",
                    items = { app.countries },
                    value = { app.countries.find((c) => c.id == app.pay.country) ?? null },
                    commit(v: object) { app.pay.country = (v as Country | null)?.id ?? "" }
                    ]
                ],
            StackedField [ label = "Postal code",
                TextInput [ width = 100%, name = "postalCode", purpose = postalCode, required = "Enter the postal code.",
                    value = { app.pay.postalCode },
                    commit(v: string) { app.pay.postalCode = tidyPostal(v, app.pay.country) },
                    validate(v: string) -> string { return postalProblem(v, app.pay.country) }
                    ]
                ]
            ]
        ]
    ]

// ── A ticket ────────────────────────────────────────────────────────────────
//
// Made to be held up at a door: the seat is the biggest thing on it, then
// when to arrive, then the code to scan.

class TicketCode [ width = 168, height = 168,
    code: string = "",
    accessibilityRole = image,
    accessibilityLabel = { "Code to scan, " + classroot.code },
    draw(d: DrawContext) {
        const n = 25
        const s = width / n
        const bits = codeBits(code, n)
        d.fillStyle = 0x1E1A17
        const finder = (cx: number, cy: number) => {
            d.fillRect(cx * s, cy * s, 7 * s, 7 * s)
            d.clearRect((cx + 1) * s, (cy + 1) * s, 5 * s, 5 * s)
            d.fillRect((cx + 2) * s, (cy + 2) * s, 3 * s, 3 * s)
            }
        for (let y = 0; y < n; y++) {
            for (let x = 0; x < n; x++) {
                const inFinder = (x < 8 && y < 8) || (x >= n - 8 && y < 8) || (x < 8 && y >= n - 8)
                if (!inFinder && bits[y * n + x]) d.fillRect(x * s, y * s, s + 0.3, s + 0.3)
                }
            }
        finder(0, 0); finder(n - 7, 0); finder(0, n - 7)
        }
    ]
