/*
# Murmur

A conversation app: six conversations, forty-four thousand messages of
history, and a live feed that keeps adding to them while you read.

How it is put together:

- **The records live in one place**, `app.murmur`, a `DataSource` class that
  loads the whole history once and owns every write to it: what arrives on the
  live socket, what you send, what you react with, what you have read. Nothing
  else holds a copy.
- **Where you are** is one attribute, `location` — the open conversation, so
  Back on a phone closes it. Everything else on the App is derived.
- **Each conversation is its own scrolling pane**, built once and kept, so
  coming back to one finds it exactly where you left it. Its messages are
  virtualized: only the rows near the screen exist. The pane reads from the
  bottom (`scrollAnchor = end`), which is the whole of "a new message comes
  into view when you are at the end, and leaves you alone when you are not".
- **A message row decides its own furniture** from its neighbours: whether it
  starts a new day, whether it continues the run of the person before it,
  whether its time is shown. The records stay the records.

The look — palettes, icons, the avatar — is in `murmur-parts.declare`.
*/

include [ "murmur-parts.declare" ]

// ── Helpers ─────────────────────────────────────────────────────────────────
//
// Functions of their arguments: dates, durations, initials, the reaction
// tally. A constraint that calls one depends only on what it passes.

script {
    type Frame = { t: string, thread?: string, person?: string, state?: string, ref?: string,
        message?: any, emoji?: string, through?: string }

    type Chip = { id: string, emoji: string, count: number, mine: boolean, who: string[] }

    const DAY = 86400000
    const timeFmt = new Intl.DateTimeFormat(undefined, { hour: "numeric", minute: "2-digit" })
    const weekdayFmt = new Intl.DateTimeFormat(undefined, { weekday: "long" })
    const shortFmt = new Intl.DateTimeFormat(undefined, { day: "numeric", month: "short" })
    const longFmt = new Intl.DateTimeFormat(undefined, { weekday: "long", day: "numeric", month: "long" })
    const yearFmt = new Intl.DateTimeFormat(undefined, { day: "numeric", month: "long", year: "numeric" })
    const listYearFmt = new Intl.DateTimeFormat(undefined, { day: "numeric", month: "short", year: "2-digit" })
    const monthYearFmt = new Intl.DateTimeFormat(undefined, { month: "long", year: "numeric" })

    function midnight(t: number): number { const d = new Date(t); d.setHours(0, 0, 0, 0); return d.getTime() }
    function daysAgo(iso: string, now: number): number {
        return Math.round((midnight(now) - midnight(Date.parse(iso))) / DAY)
    }
    function dayKey(iso: string): string { const d = new Date(iso); return d.getFullYear() + "-" + d.getMonth() + "-" + d.getDate() }
    function sameRun(a: string, b: string): boolean { return Math.abs(Date.parse(b) - Date.parse(a)) < 6 * 60000 }
    function hhmm(iso: string): string { return iso == "" ? "" : timeFmt.format(new Date(iso)) }

    // A day heading: Today, Yesterday, a weekday this week, else the date.
    function dayLabel(iso: string, now: number): string {
        if (iso == "") return ""
        const n = daysAgo(iso, now)
        const d = new Date(iso)
        if (n == 0) return "Today"
        if (n == 1) return "Yesterday"
        if (n < 7 && n > 0) return weekdayFmt.format(d)
        return d.getFullYear() == new Date(now).getFullYear() ? longFmt.format(d) : yearFmt.format(d)
    }

    // The list's time: a clock time today, a weekday this week, else a date.
    function stamp(iso: string, now: number): string {
        if (iso == "") return ""
        const n = daysAgo(iso, now)
        const d = new Date(iso)
        if (n == 0) return timeFmt.format(d)
        if (n == 1) return "Yesterday"
        if (n < 7 && n > 0) return weekdayFmt.format(d).slice(0, 3)
        return d.getFullYear() == new Date(now).getFullYear() ? shortFmt.format(d) : listYearFmt.format(d)
    }

    function since(iso: string): string { return iso == "" ? "" : monthYearFmt.format(new Date(iso)) }

    function clock(sec: number): string {
        const s = Math.max(0, Math.round(sec))
        return Math.floor(s / 60) + ":" + String(s % 60).padStart(2, "0")
    }

    function initials(name: string): string {
        const parts = name.trim().split(/\s+/).filter((p) => p != "")
        if (parts.length == 0) return ""
        return (parts[0][0] + (parts.length > 1 ? parts[parts.length - 1][0] : "")).toUpperCase()
    }

    function grouped(n: number): string { return n.toLocaleString() }

    // Three dots, each brightening in turn.
    function dotLevel(phase: number, i: number): number {
        const p = (phase / 1.2 - i * 0.18 + 1) % 1
        return 0.3 + 0.7 * Math.max(0, Math.sin(p * Math.PI * 1.6))
    }

    // Reactions as the chips under a message: one per mark, how many, whether
    // one of them is yours, and who.
    function tally(list: { person: string, emoji: string }[], me: string): Chip[] {
        const out: Chip[] = []
        for (const r of list) {
            let c = out.find((x) => x.emoji == r.emoji)
            if (!c) { c = { id: r.emoji, emoji: r.emoji, count: 0, mine: false, who: [] }; out.push(c) }
            c.count += 1
            c.who.push(r.person)
            if (r.person == me) c.mine = true
        }
        return out
    }
    }


// ── The shape of the data ───────────────────────────────────────────────────

schema Person [ id: string, name: string, avatar?: string ]
schema Sendable [ src: string, alt: string ]
schema Reaction [ person: string, emoji: string ]

// A message as the service sends it, plus what this app adds to one you sent:
// `pending` until the service answers, then its `sid`, the id it was given.

schema Message [ id: string, from: string, at: string, kind: "text" | "photo" | "voice",
    text?: string, src?: string, alt?: string, durationMs?: number, peaks[]?: number,
    reactions[]?: Reaction, pending?: boolean, sid?: string ]

// Who has read how far, as an index into the messages.

schema Seen [ person: string, index: number ]

// A conversation as served, plus what this session knows about it: how far
// you have read (`read`, a count of messages), where the unread ones began
// when you last opened it (`mark`), who is writing now, and who has seen what.

schema Thread [ id: string, title?: string, participants[]: string, lastSeen: string,
    messages[]: Message, read?: number, mark?: number, typing[]?: string, seen[]?: Seen ]

schema Inbox [ rows[]: Thread ]
schema Emoji [ id: string, e: string ]


// ── The records ─────────────────────────────────────────────────────────────
//
// The whole history arrives once. The live socket opens when it has, and
// every frame it brings is written into the same records — so the list, the
// open conversation and the unread counts all follow with no code of theirs.

class Murmur extends DataSource [ autoFetch = true,
    schema = [ me: string, people[]: Person, sendable[]: Sendable, threads[]: Thread ],
    base: string = "",
    url = { classroot.base == "" ? "" : classroot.base + "/threads.json" },

    me: string = { this.value?.me ?? "" },
    sent: number = 0,                       // how many you have sent: each gets its own ref

    // ── reading

    people() -> Person[] { return this.value?.people ?? [] },
    threads() -> Thread[] { return this.value?.threads ?? [] },
    at(id: string) -> number { return this.threads().findIndex((t) => t.id == id) },
    thread(id: string) -> Thread? { return this.threads().find((t) => t.id == id) ?? null },
    get(id: string) -> Thread { return this.thread(id) ?? ({ id: id, participants: [], lastSeen: "", messages: [] }) },
    person(id: string) -> Person? { return this.people().find((p) => p.id == id) ?? null },
    nameOf(id: string) -> string { return id == this.me ? "You" : this.person(id)?.name ?? "" },
    firstName(id: string) -> string { return id == this.me ? "You" : this.nameOf(id).split(" ")[0] },
    hueOf(id: string) -> number { return Math.max(0, this.people().findIndex((p) => p.id == id)) % 8 },
    media(path: string) -> string { return path == "" ? "" : this.base + "/" + path },
    avatarOf(id: string) -> string { return this.media(this.person(id)?.avatar ?? "") },
    altOf(src: string) -> string { return (this.value?.sendable ?? []).find((s) => s.src == src)?.alt ?? "" },

    others(t: Thread) -> string[] { return t.participants.filter((p) => p != this.me) },
    titleIn(tid: string) -> string {
        const t = this.thread(tid)
        if (t == null) return ""
        if (t.title != null) return t.title
        const others = t.participants.filter((p) => p != this.me)
        return others.length == 1 ? this.nameOf(others[0]) : others.map((p) => this.firstName(p)).join(", ")
        },
    lastOf(t: Thread) -> Message? { return t.messages[t.messages.length - 1] ?? null },
    lastAt(t: Thread) -> string { return this.lastOf(t)?.at ?? "" },

    // The faces a conversation shows: the most recent other voices first.
    faces(t: Thread) -> string[] {
        const out: string[] = []
        for (let i = t.messages.length - 1; i >= 0 && out.length < 2 && i > t.messages.length - 60; i--) {
            const f = t.messages[i].from
            if (f != this.me && !out.includes(f)) out.push(f)
            }
        for (const p of this.others(t)) if (out.length < 2 && !out.includes(p)) out.push(p)
        return out
        },

    // What has arrived from others since you last looked.
    unread(t: Thread) -> number {
        let n = 0
        for (let i = t.read ?? t.messages.length; i < t.messages.length; i++) if (t.messages[i].from != this.me) n++
        return n
        },
    // The same, by conversation id: what the views ask.
    facesIn(tid: string) -> string[] { return this.faces(this.get(tid)) },
    unreadIn(tid: string) -> number { return this.unread(this.get(tid)) },
    lastAtIn(tid: string) -> string { return this.lastAt(this.get(tid)) },
    othersIn(tid: string) -> string[] { return this.others(this.get(tid)) },
    totalUnread: number = { this.threads().reduce((n, t) => n + this.unread(t), 0) },

    // What a message says, in a line: for the list, and for the reaction sheet.
    gist(m: Message?) -> string {
        if (m == null) return ""
        if (m.kind == "photo") return "Photo"
        if (m.kind == "voice") return "Voice message · " + clock((m.durationMs ?? 0) / 1000)
        return m.text ?? ""
        },
    previewIn(tid: string) -> string { return this.preview(this.get(tid)) },
    preview(t: Thread) -> string {
        const m = this.lastOf(t)
        if (m == null) return "No messages yet"
        const who = m.from == this.me ? "You: " : this.others(t).length > 1 ? this.firstName(m.from) + ": " : ""
        return who + this.gist(m)
        },

    // The conversations, most recent first. These are the threads themselves,
    // not copies, so the list's rows read the live records.
    byRecent() -> Inbox {
        const rows = [...this.threads()]
        rows.sort((a, b) => this.lastAt(b).localeCompare(this.lastAt(a)))
        return ({ rows: rows })
        },

    // Where a message is in its conversation, searching from the end, where
    // anything that changes almost always is.
    find(i: number, id: string) -> number {
        const ms = this.threads()[i]?.messages ?? []
        for (let j = ms.length - 1; j >= 0; j--) if (ms[j].id == id || ms[j].sid == id) return j
        return -1
        },

    // ── arrival: everything the service says is written here

    onLoad() {
        this.threads().forEach((t, i) => {
            const j = t.messages.findIndex((m) => m.id == t.lastSeen)
            this.set(["threads", i, "read"], j + 1)
            })
        },

    live: Socket [ url = { classroot.loaded ? classroot.base.replace(/^http/, "ws") + "/live" : "" },
        onMessage(e: StreamMessage) { classroot.hear(JSON.parse(e.data) as Frame) }
        ],

    hear(f: Frame) {
        const i = this.at(f.thread ?? "")
        if (f.t == "sent") { this.confirm(f.ref ?? "", f.message as Message); return }
        if (i < 0) return
        if (f.t == "message") this.arrive(i, f.message as Message)
        else if (f.t == "reaction") this.attach(i, f.message as string, f.person ?? "", f.emoji ?? "")
        else if (f.t == "composing") this.composing(i, f.person ?? "", f.state == "start")
        else if (f.t == "seen") this.seenBy(i, f.person ?? "", f.through ?? "")
        },

    arrive(i: number, m: Message) {
        this.composing(i, m.from, false)
        this.set(["threads", i, "messages", "-"], m)
        },

    composing(i: number, person: string, on: boolean) {
        const now = this.threads()[i].typing ?? []
        if (on == now.includes(person)) return
        this.set(["threads", i, "typing"], on ? [...now, person] : now.filter((p) => p != person))
        },

    seenBy(i: number, person: string, through: string) {
        const j = this.find(i, through)
        if (j < 0) return
        const now = (this.threads()[i].seen ?? []).filter((s) => s.person != person)
        this.set(["threads", i, "seen"], [...now, ({ person: person, index: j })])
        },

    attach(i: number, id: string, person: string, emoji: string) {
        const j = this.find(i, id)
        if (j < 0) return
        const now = this.threads()[i].messages[j].reactions ?? []
        if (now.some((r) => r.person == person && r.emoji == emoji)) return
        this.set(["threads", i, "messages", j, "reactions"], [...now, ({ person: person, emoji: emoji })])
        },

    // The service's answer to something you sent: it keeps its place and
    // identity here, and learns the id the service gave it.
    confirm(ref: string, m: Message) {
        this.threads().forEach((t, i) => {
            const j = this.find(i, ref)
            if (j < 0) return
            this.set(["threads", i, "messages", j, "sid"], m.id)
            this.set(["threads", i, "messages", j, "at"], m.at)
            this.set(["threads", i, "messages", j, "pending"], false)
            })
        },

    // ── what you do

    send(tid: string, kind: "text" | "photo", body: string) {
        const i = this.at(tid)
        if (i < 0 || body.trim() == "") return
        this.sent = this.sent + 1
        const ref = "local-" + this.sent
        const at = new Date().toISOString()
        const m: Message = kind == "text"
            ? ({ id: ref, from: this.me, at: at, kind: "text", text: body, pending: true })
            : ({ id: ref, from: this.me, at: at, kind: "photo", src: body, alt: this.altOf(body), pending: true })
        this.set(["threads", i, "messages", "-"], m)
        this.set(["threads", i, "read"], this.threads()[i].messages.length + 1)
        this.live.send(JSON.stringify(kind == "text"
            ? ({ t: "send", ref: ref, thread: tid, kind: "text", text: body })
            : ({ t: "send", ref: ref, thread: tid, kind: "photo", src: body })))
        },

    react(tid: string, id: string, emoji: string) {
        const i = this.at(tid)
        const j = this.find(i, id)
        if (j < 0) return
        const m = this.threads()[i].messages[j]
        this.attach(i, id, this.me, emoji)
        this.live.send(JSON.stringify(({ t: "react", thread: tid, message: m.sid ?? m.id, emoji: emoji })))
        },

    // Opening a conversation remembers where its unread messages began, for
    // the "New" rule, then counts them read.
    opened(tid: string) {
        const i = this.at(tid)
        if (i < 0) return
        const t = this.threads()[i]
        const r = t.read ?? t.messages.length
        this.set(["threads", i, "mark"], this.unread(t) > 0 ? r : -1)
        this.markRead(tid)
        },
    markRead(tid: string) {
        const i = this.at(tid)
        if (i < 0) return
        const n = this.threads()[i].messages.length
        if ((this.threads()[i].read ?? 0) != n) this.set(["threads", i, "read"], n)
        }
    ]


// ── The arrangement ─────────────────────────────────────────────────────────
//
// Two children: the list and the open conversation. On a wide screen they
// sit side by side. On a phone the conversation is pushed over the list, and
// `push` (sprung, 0 to 1) is how far: the list slides a little way left and
// dims as the conversation comes in over it, as a stack of pages does.

class Stage extends Layout [
    wide: boolean = true,
    side: number = 360,
    push: number = 0,
    place() {
        const w = this.contentExtent("width")
        const h = this.contentExtent("height")
        const p = this.push
        return this.laid().map((c, i) => this.wide
            ? (i == 0 ? ({ x: 0, y: 0, w: this.side, h: h }) : ({ x: this.side, y: 0, w: w - this.side, h: h }))
            : (i == 0 ? ({ x: -0.28 * w * p, y: 0, w: w, h: h, vis: p < 0.999 })
                      : ({ x: w * (1 - p), y: 0, w: w, h: h, vis: p > 0.001 })))
        }
    ]


// The list's rows, stacked; when the order changes they glide to their new
// places instead of jumping, so a conversation that rises to the top is seen
// to rise.

class Stack extends TweenLayout [ duration = 380,
    gap: number = 2,
    place() {
        const w = this.contentExtent("width")
        let y = 0
        return this.laid().map((c) => {
            const box = ({ x: 0, y: y, w: w, h: c.height })
            y = y + c.height + this.gap
            return box
            })
        }
    ]

// ── A message ───────────────────────────────────────────────────────────────
//
// Every kind of message has the same furniture around it, so the base class
// draws it and the kinds supply only their bubble (`defaultPlacement`). What
// the furniture shows is decided from the neighbours: the list provides its
// records (`rows`), and a row reads the one before and after it.
//
//   a day heading      — when this message starts a day
//   the "New" rule     — when the unread messages began here
//   the sender's name  — in a group, at the start of someone's run
//   the bubble         — corners tightened where the run continues
//   its reactions
//   its time           — at the end of a run, or "Sending…", or "Seen"

class MsgRow [ width = 100%, defaultPlacement = slot,
    mid: string = { :id ?? "" },
    from: string = { :from ?? "" },
    when: string = { :at ?? "" },
    prev: Message? = { rowIndex > 0 ? provided("rows")[rowIndex - 1] ?? null : null },
    next: Message? = { provided("rows")[rowIndex + 1] ?? null },
    mine: boolean = { classroot.from == app.murmur.me },
    group: boolean = { provided("group") },
    newDay: boolean = { classroot.prev == null || dayKey(classroot.prev.at) != dayKey(classroot.when) },
    startsUnread: boolean = { rowIndex == provided("mark") },
    nextStartsUnread: boolean = { rowIndex + 1 == provided("mark") },
    sameDayAsNext: boolean = { classroot.next != null && dayKey(classroot.next.at) == dayKey(classroot.when) },
    prevIsMine: boolean = { classroot.prev != null && classroot.prev.from == classroot.from },
    nextIsMine: boolean = { classroot.next != null && classroot.next.from == classroot.from },
    soonAfterPrev: boolean = { classroot.prev != null && sameRun(classroot.prev.at, classroot.when) },
    soonBeforeNext: boolean = { classroot.next != null && sameRun(classroot.when, classroot.next.at) },
    runsOn: boolean = { classroot.prevIsMine && classroot.soonAfterPrev },
    runsInto: boolean = { classroot.nextIsMine && classroot.soonBeforeNext },
    joinPrev: boolean = { classroot.runsOn && !classroot.newDay && !classroot.startsUnread },
    joinNext: boolean = { classroot.runsInto && classroot.sameDayAsNext && !classroot.nextStartsUnread },
    showName: boolean = { classroot.group && !classroot.mine && !classroot.joinPrev },
    pending: boolean = { :pending ?? false },

    // The paper this message is set on, and its ink: yours in deep ink, each
    // other person's on their own tint.
    hue: number = { app.murmur.hueOf(classroot.from) },
    paper: Color = { classroot.mine ? provided("theme").mine : provided("theme").tints[classroot.hue] },
    ink: Color = { classroot.mine ? provided("theme").mineText : provided("theme").text },
    big: number = 20,
    small: number = 6,
    corners: Radius = { classroot.mine
        ? [classroot.big, classroot.joinPrev ? classroot.small : classroot.big, classroot.joinNext ? classroot.small : classroot.big, classroot.big]
        : [classroot.joinPrev ? classroot.small : classroot.big, classroot.big, classroot.big, classroot.joinNext ? classroot.small : classroot.big] },

// The scrubber. The recording's loudness, drawn as bars; what has played is
// solid, what is still to come is faint, and the point reached is a bead.
// Press anywhere on it to go there; drag sideways to move through it (a
// vertical drag still scrolls the conversation). It is a slider to a screen
// reader, and the arrow keys step five seconds.

class Scrubber extends Control [ claim = x,
    peaks: number[] = [],
    secs: number = 0,
    at: number = 0,
    live: boolean = false,
    ink: Color = gray,
    dragging: boolean = false,
    frac: number = { classroot.secs > 0 ? Math.min(1, classroot.at / classroot.secs) : 0 },

    scrub(frac: number) { },
    letGo() { },
    scrubAt(x: number) { scrub(Math.max(0, Math.min(1, x / Math.max(1, width)))) },

    accessibilityRole = slider,
    accessibilityValue = { Math.round(at) },
    accessibilityMin = 0,
    accessibilityMax = { Math.round(secs) },
    accessibilityIncrement() { scrub(Math.min(1, (at + 5) / Math.max(secs, 0.1))); letGo() },
    accessibilityDecrement() { scrub(Math.max(0, (at - 5) / Math.max(secs, 0.1))); letGo() },

    onPointerDown(e: PointerEvent) { super.onPointerDown(); dragging = true; scrubAt(e.x) },
    onPointerMove(e: PointerEvent) { if (dragging && pressed) scrubAt(e.x) },
    onPointerUp(e: PointerUpEvent) { if (dragging) { dragging = false; letGo() } },

    // As many bars as fit, each the loudest of the samples it covers.
    draw(d: DrawContext) {
        if (peaks.length == 0) return
        const w = d.w, h = d.h
        const bar = 2.5, gap = 2
        const n = Math.max(1, Math.floor((w + gap) / (bar + gap)))
        const step = (w + gap) / n
        const cut = frac * w
        const shown = live || dragging
        d.fillStyle = ink
        for (let i = 0; i < n; i++) {
            const a = Math.floor(i * peaks.length / n), b = Math.max(a + 1, Math.floor((i + 1) * peaks.length / n))
            const level = Math.max(...peaks.slice(a, b))
            const x = i * step
            const bh = Math.max(3, level * (h - 6))
            d.globalAlpha = shown && x + bar / 2 <= cut ? 1 : 0.4
            d.beginPath()
            d.roundRect(x, (h - bh) / 2, bar, bh, bar / 2)
            d.fill()
            }
        d.globalAlpha = 1
        if (shown) {
            d.beginPath()
            d.arc(Math.max(5, Math.min(w - 5, cut)), h / 2, dragging ? 7 : 5, 0, Math.PI * 2)
            d.fill()
            }
        }
    ]

    resume: boolean = false,
    load(id: string, src: string, secs: number) {
        if (voiceId == id) return
        voiceId = id
        voiceSrc = murmur.media(src)
        voiceSecs = secs
        head = 0
        },
    toggleVoice(id: string, src: string, secs: number) {
        if (voiceId == id) { voicePlaying = !voicePlaying; return }
        load(id, src, secs)
        voicePlaying = true
        },
    seekVoice(id: string, src: string, secs: number, frac: number) {
        if (voiceId != id) { load(id, src, secs); resume = true }
        else if (!resume) resume = voicePlaying
        voicePlaying = false
        head = frac * secs
        app.convo.player.position = head
        },
    endScrub() { if (resume) voicePlaying = true; resume = false },

    // ── the typing dots' shared phase, running only while someone types

    phase: number = 0,
    anyTyping: boolean = { app.murmur.threads().some((t) => (t.typing ?? []).length > 0) },
    dots: Time [ tick = frame, running = { app.anyTyping }, onTick(dt: number) { app.phase = (app.phase + dt / 1000) % 1.2 } ],

                    text = { !app.murmur.loaded ? "" : app.murmur.totalUnread > 0 ? app.murmur.totalUnread + " new since you last looked" : "All caught up" } ],
                Text [ fontSize = 14, textColor = { provided("theme").textFaint }, text = "Choose a conversation to read" ]
                ]
            ],

        player: Audio [ source = { app.voiceSrc }, playing = { app.voicePlaying },
            trackChanges = ["ended"],
            onChange() { if (this.ended) { app.voicePlaying = false; app.head = 0 } }
            ],
        playhead: Time [ tick = frame, running = { app.voicePlaying },
            onTick(dt: number) {
                const p = app.convo.player.position
                const next = app.head + dt / 1000
                app.head = Math.abs(next - p) > 0.3 ? p : Math.min(next, app.voiceSecs)
                }
            ]
        ],

    splash: View [ ignoreLayout = true, visible = { !app.murmur.loaded }, width = { app.width }, height = { app.height },
