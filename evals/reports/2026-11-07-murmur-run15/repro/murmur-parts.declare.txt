/*
# Murmur: the parts

The look of Murmur, kept apart from how it works: the two palettes, the
drawn icons, and the small pieces every screen is built from (an avatar, an
icon button, the typing dots). `murmur.declare` includes this file.

The one idea in the palette: **every person has a colour, and the colour is
the attribution.** A person's `hue` indexes three parallel lists — `fills`
(their avatar, the dots when they type), `inks` (their name) and `tints` (the
paper their words are set on). You are always the same deep ink.
*/

// ── Icons ───────────────────────────────────────────────────────────────────
//
// One pen for every mark: a round-ended stroke on the library's 16-unit grid.

class Glyph extends Icon [
    shape: string = "",
    draw(d: DrawContext) {
        d.scale(unit, unit)
        d.strokeStyle = ink
        d.fillStyle = ink
        d.lineWidth = weight / unit
        d.lineCap = "round"
        d.lineJoin = "round"
        d.beginPath()
        if (shape == "back") {
            d.moveTo(10.5, 2.5); d.lineTo(5, 8); d.lineTo(10.5, 13.5); d.stroke()
        } else if (shape == "send") {                       // an arrow up
            d.moveTo(8, 13.5); d.lineTo(8, 2.8); d.moveTo(3.2, 7.4); d.lineTo(8, 2.6); d.lineTo(12.8, 7.4); d.stroke()
        } else if (shape == "photo") {                      // a frame with a sun and a hill
            d.roundRect(1.8, 3, 12.4, 10, 2)
            d.moveTo(1.8, 11); d.lineTo(6, 7.4); d.lineTo(9.2, 10.2); d.lineTo(11, 8.8); d.lineTo(14.2, 11.4)
            d.stroke(); d.beginPath(); d.arc(10.6, 6.2, 1.1, 0, 6.3); d.fill()
        } else if (shape == "play") {
            d.moveTo(5, 3); d.lineTo(13, 8); d.lineTo(5, 13); d.closePath(); d.fill(); d.stroke()
        } else if (shape == "pause") {
            d.roundRect(4, 3, 2.6, 10, 0.8); d.roundRect(9.4, 3, 2.6, 10, 0.8); d.fill()
        } else if (shape == "down") {
            d.moveTo(8, 2.8); d.lineTo(8, 13); d.moveTo(3.4, 8.6); d.lineTo(8, 13.2); d.lineTo(12.6, 8.6); d.stroke()
        } else if (shape == "close") {
            d.moveTo(3.5, 3.5); d.lineTo(12.5, 12.5); d.moveTo(12.5, 3.5); d.lineTo(3.5, 12.5); d.stroke()
        } else if (shape == "react") {                      // a face with a small plus
            d.arc(7, 9, 5.6, 0, 6.3)
            d.moveTo(4.6, 10.6); d.quadraticCurveTo(7, 12.8, 9.4, 10.6)
            d.moveTo(12.6, 1.2); d.lineTo(12.6, 5.4); d.moveTo(10.5, 3.3); d.lineTo(14.7, 3.3)
            d.stroke(); d.beginPath(); d.arc(5.2, 7.6, 0.75, 0, 6.3); d.arc(8.8, 7.6, 0.75, 0, 6.3); d.fill()
        }
        }
    ]


// ── An avatar ───────────────────────────────────────────────────────────────
//
// A person's picture in a circle, or — when they have none, or it has not
// arrived yet — their initials on their own colour. The picture lies over the
// initials and fades in when it lands, so nothing pops.

class Avatar [ width = 32, height = { this.width }, cornerRadius = { this.width / 2 }, clip = true,
    person: string = "",
    hue: number = { app.murmur.hueOf(classroot.person) },
    src: string = { app.murmur.avatarOf(classroot.person) },
    fill = { provided("theme").fills[classroot.hue] },
    accessibilityRole = image,
    accessibilityLabel = { app.murmur.nameOf(classroot.person) },
    TextLabel [ x = center, y = center, fontSize = { classroot.width * 0.38 }, fontWeight = semibold,
        letterSpacing = 0.3, textColor = white, text = { initials(app.murmur.nameOf(classroot.person)) } ],
    pic: Image [ exists = { classroot.src != "" }, width = 100%, height = 100%, stretches = cover,
        source = { classroot.src }, opacity = { this.loaded ? 1 : 0 } ]
    ]


// A conversation's face in the list: the other person, or for a group the two
// most recent other voices, one tucked behind the other.

class ThreadFace [ width = { classroot.side }, height = { classroot.side },
    side: number = 52,
    faces: string[] = [],
    small: number = { Math.round(classroot.side * 0.7) },
    one: Avatar [ exists = { classroot.faces.length < 2 }, width = { classroot.side }, person = { classroot.faces[0] ?? "" } ],
    back: Avatar [ exists = { classroot.faces.length >= 2 }, width = { classroot.small }, person = { classroot.faces[1] ?? "" } ],
    front: Avatar [ exists = { classroot.faces.length >= 2 }, x = { classroot.side - classroot.small }, y = { classroot.side - classroot.small },
        width = { classroot.small }, stroke = { stroke(2.5, provided("theme").rail) }, person = { classroot.faces[0] ?? "" } ]
    ]


// ── Small controls ──────────────────────────────────────────────────────────

// A round icon button: back, photo, send, close.

class IconButton extends Control [ width = 44, height = 44, cornerRadius = 22,
    shape: string = "",
    tone: Color = gray,
    size: number = 18,
    fill = { pressed ? provided("theme").controlPressed : hovered ? provided("theme").hover : null },
    Glyph [ x = center, y = center, iconSize = { classroot.size }, weight = 1.8,
        shape = { classroot.shape }, ink = { classroot.tone } ]
    ]


// Three dots that breathe, in the colour of whoever is writing. A frame clock
// runs only while some exist; each dot reads the shared phase.

class TypingDots [ width = 30, height = 10,
    tone: Color = gray,
    layout: SimpleLayout [ axis = x, spacing = 4, align = center ],
    View [ width = 6, height = 6, cornerRadius = 3, fill = { classroot.tone }, opacity = { dotLevel(app.phase, 0) } ],
    View [ width = 6, height = 6, cornerRadius = 3, fill = { classroot.tone }, opacity = { dotLevel(app.phase, 1) } ],
    View [ width = 6, height = 6, cornerRadius = 3, fill = { classroot.tone }, opacity = { dotLevel(app.phase, 2) } ]
    ]
