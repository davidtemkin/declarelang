/*
# The Venue

The box office for *The Marriage of Figaro* in the Grand Hall: pick a
performance, choose seats in the hall, pay, and come away with tickets to
show at the door. On a phone and at a desk, one program.

How it is put together:

- **Everything lives behind the service** (`http://127.0.0.1:8461`). Each thing
  it holds is a `DataSource` on the App: the production, the hall's seats
  (`Hall`, a class that also knows how the house is laid out), who is signed
  in, and the three writes — signing in, signing out, and the order. A stream
  (`feed`) reports seats other people buy, and writes them into the hall.
- **Where the person is** is a few attributes on the App: the performance is
  the `location` (an address you could hand someone), the step past the hall
  is the `waypoint` (Back undoes it, a stranger never sees it), and the seats
  chosen are a list of ids.
- **What they type at checkout lives on the App too**, not in the form, so
  signing in halfway through loses nothing. The fields that start from the
  account (email, name, billing address, which card) are computed defaults:
  they follow the account until the person types, and then they are theirs.
- **The form is a `Form`** sending the order. It judges every field before
  anything is sent, sends once however often Pay is pressed, and lays the
  service's refusals on the field they name.

The look and the small parts (a seat, the hall's floor, a ticket) are in
`venue-parts.declare`.
*/

include [ "venue-parts.declare" ]

// ── The hall ────────────────────────────────────────────────────────────────
//
// The seat map for one performance, and how the house stands: rows curve
// round the stage, front to back, with a centre aisle; each tier is set back
// from the one before. `laid` wraps every seat record with its place, in seat
// pitches; the views multiply by the pitch they are drawn at.

class Hall extends DataSource [ schema = SeatMap, credentials = include, autoFetch = true,

    sections() -> Section[] { return this.value?.sections ?? [] },
    laid: Dataset [ schema = Laid, contents = { classroot.arrange() } ],

    arrange() -> Laid {
        const R0 = 20, ROW = 1.0, GAP = 2.7, AISLE = 0.9, M = 0.9
        const seats: Placed[] = [], floors: Floor[] = [], rows: RowMark[] = []
        let r = R0
        this.sections().forEach((sec, ti) => {
            const first = r
            let half = 0
            sec.rows.forEach((row) => {
                const n = row.seats.length
                row.seats.forEach((s, k) => {
                    const along = k - (n - 1) / 2 + (k >= n / 2 ? AISLE / 2 : -AISLE / 2)
                    const th = along / r
                    seats.push(({ id: s.id, seat: s, label: sec.name + ", row " + row.name + ", seat " + s.number,
                        section: sec.name, row: row.name, number: s.number, tier: ti,
                        priceCents: sec.priceCents, feeCents: sec.feeCents,
                        ux: r * Math.sin(th), uy: r * Math.cos(th) }))
                    })
                const edge = ((n - 1) / 2 + AISLE / 2 + 1.05) / r
                rows.push(({ id: sec.key + "-" + row.name, name: row.name, ux: -r * Math.sin(edge), uy: r * Math.cos(edge) }))
                half = Math.max(half, edge + 0.7 / r)
                r = r + ROW
                })
            floors.push(({ key: sec.key, name: sec.name, priceCents: sec.priceCents, tier: ti,
                r0: first - 0.75, r1: r - ROW + 0.75, half: half }))
            r = r - ROW + GAP
            })
        if (seats.length == 0) return ({ seats: [], floors: [], rows: [], cx: 0, cy: 0, w: 1, h: 1 })

        // the box round it all, the stage included
        const s0 = floors[0].r0
        let minX = 0, maxX = 0, minY = s0 - 4.2, maxY = 0
        floors.forEach((f) => {
            minX = Math.min(minX, -f.r1 * Math.sin(f.half))
            maxX = Math.max(maxX, f.r1 * Math.sin(f.half))
            minY = Math.min(minY, f.r0 * Math.cos(f.half))
            maxY = Math.max(maxY, f.r1)
            })
        const dx = M - minX, dy = M - minY
        return ({
            seats: seats.map((s) => ({ ...s, ux: s.ux + dx, uy: s.uy + dy })),
            rows: rows.map((m) => ({ ...m, ux: m.ux + dx, uy: m.uy + dy })),
            floors: floors, cx: dx, cy: dy, w: maxX - minX + 2 * M, h: maxY - minY + 2 * M })
        },

    tierRows() -> TierRow[] {
        return this.sections().map((sec) => ({ id: sec.key, name: sec.name, priceCents: sec.priceCents,
            free: sec.rows.reduce((n, row) => n + row.seats.filter((s) => s.status == "available").length, 0) }))
        },
    tiers: Dataset [ schema = [ rows[]: TierRow ], contents = { { rows: classroot.tierRows() } } ],

    // The free seat in the next row forward or back that sits nearest
    // across: where Up and Down take the keyboard in the hall.

    neighbour(id: string, step: number) -> string {
        const all = this.laid.value?.seats ?? []
        const at = all.find((s) => s.id == id)
        if (at == null) return ""
        const rows: string[] = []
        all.forEach((s) => { const k = s.tier + ":" + s.row; if (!rows.includes(k)) rows.push(k) })
        for (let r = rows.indexOf(at.tier + ":" + at.row) + step; r >= 0 && r < rows.length; r = r + step) {
            const near = all.filter((s) => s.tier + ":" + s.row == rows[r] && s.seat.status == "available")
                .sort((a, b) => Math.abs(a.ux - at.ux) - Math.abs(b.ux - at.ux))
            if (near.length > 0) return near[0].id
            }
        return ""
        },

    placed(id: string) -> Placed? { return (this.laid.value?.seats ?? []).find((s) => s.id == id) ?? null },
    free: number = { classroot.sections().reduce((n, sec) => n + sec.rows.reduce((m, row) =>
        m + row.seats.filter((s) => s.status == "available").length, 0), 0) },
    total: number = { classroot.sections().reduce((n, sec) => n + sec.rows.reduce((m, row) => m + row.seats.length, 0), 0) },

    // Someone else bought it: the seat's own record changes, and every view
    // of it follows.

    markSold(id: string) {
        this.sections().forEach((sec, si) => sec.rows.forEach((row, ri) => row.seats.forEach((s, k) => {
            if (s.id == id && s.status != "sold") this.set(["sections", si, "rows", ri, "seats", k, "status"], "sold")
            })))
        }
    ]


// ── The checkout form ───────────────────────────────────────────────────────
//
// Every value is the App's (so signing in loses nothing); the form judges them,
// sends the order, and answers the service's refusals. A field marked wrong
// says so only once it has been left, or on Pay.

class PaymentForm extends Form [ source = { app.order }, enterMoves = true,
    layout: SimpleLayout [ axis = y, spacing = 28 ],

    saved: boolean = { app.payWith != "new" && app.signedIn },
    brand: string = { cardBrand(digitsOf(app.cardNumber)) },

    // A refusal the form can't lay on a field, said in the program's words.

    failedMessage = "The payment couldn't be sent. Check your connection and try again.",
    uncertainMessage = "We didn't hear back about your payment. Check your email for tickets before you try again.",

    refusedWith(body: Refusal) {
        if (body.error == "unavailable") { app.seatsLost(body.conflicts ?? [], body.message); return }
        if (body.error == "card_declined" || body.error == "insufficient_funds") {
            // everything stays but the security code, which is typed again
            app.cvc = ""
            app.savedCvc = ""
            if (classroot.cvcField != null) classroot.cvcField.judged = false
            if (classroot.savedCvcField != null) classroot.savedCvcField.judged = false
            }
        },
    cvcField: TextInput? = { classroot.newCard.cardRow.cvcBox.field },
    savedCvcField: TextInput? = { classroot.payment.savedCode.field },

    // ── who the tickets go to

    contact: FormSection [ visible = { !classroot.saved }, title = "Contact",
        Field [ width = 100%, label = "Email", stacked = true,
            Entry [ name = "email", purpose = email, required = "Enter the email address to send your tickets to.",
                accessibilityHint = "Your tickets are sent here.",
                value <-> app.email,
                validate(v: string) -> string { return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v.trim()) ? "" : "Enter an email address like name@example.com." }
                ]
            ],
        Text [ width = 100%, fontSize = 13, textColor = { provided("theme").textMuted },
            text = "Your tickets are sent here. There's nothing to post and nothing to collect." ]
        ],

    // ── how they pay

    payment: FormSection [ title = "Payment",
        Text [ visible = { !app.signedIn }, width = 100%, fontSize = 14, textColor = { provided("theme").textMuted },
            text = "No account needed. If you have one, sign in at the top of the page to use a saved card." ],
        cards: RadioGroup [ visible = { app.signedIn }, width = 100%, datapath = { app.profile },
            accessibilityLabel = "Pay with", value <-> app.payWith,
            layout: SimpleLayout [ axis = y, spacing = 8 ],
            SavedCard [ datapath = :cards[] ],
            NewCardChoice [ ]
            ],
        savedCode: Field [ visible = { classroot.saved }, width = 200, label = "Security code", stacked = true,
            field: Entry [ name = "cvc", purpose = cardCode,
                required = "Enter the security code from the back of your card.",
                value <-> app.savedCvc,
                admit(t: string) -> string { return digitsOf(t).slice(0, 4) },
                validate(v: string) -> string { return digitsOf(v).length == codeLength(app.savedBrand) ? "" : "Enter the " + codeLength(app.savedBrand) + "-digit code from your card." }
                ]
            ]
        ],

    newCard: FormSection [ visible = { !classroot.saved }, title = { app.signedIn ? "New card" : "Card" },
        Field [ width = 100%, label = "Name on card", stacked = true,
            Entry [ name = "name", purpose = cardName, required = "Enter the name as it's printed on the card.",
                value <-> app.cardName ]
            ],
        Field [ width = 100%, label = "Card number", stacked = true,
            number: Entry [ name = "number", purpose = cardNumber, numeralWidth = tabular,
                padding = { [0, classroot.brand == "" ? 12 : 92, 0, 12] }, height = { app.touchFirst ? 44 : 40 },
                required = "Enter your card number.",
                value <-> app.cardNumber,
                admit(t: string) -> string { const d = digitsOf(t); return groupCard(d.slice(0, cardLength(cardBrand(d)))) },
                validate(v: string) -> string { return cardProblem(v) },
                brandTag: View [ x = { parent.width - this.width - 22 }, y = center, visible = { classroot.brand != "" },
                    pointerEvents = "none", accessibilityLive = polite,
                    BrandMark [ brand = { classroot.brand }, accessibilityHidden = false,
                        accessibilityLabel = { brandName(classroot.brand) } ]
                    ]
                ]
            ],
        cardRow: View [ width = 100%,
            layout: SimpleLayout [ axis = x, spacing = 12 ],
            Field [ flexes = true, label = "Expiry", stacked = true,
                Entry [ name = "expiry", purpose = cardExpiry, placeholder = "MM / YY", numeralWidth = tabular,
                    required = "Enter the expiry date from your card.",
                    value <-> app.expiry,
                    admit(t: string) -> string { return tidyExpiry(t) },
                    validate(v: string) -> string { return expiryProblem(v, app.clock.year, app.clock.month) }
                    ]
                ],
            cvcBox: Field [ flexes = true, label = "Security code", stacked = true,
                field: Entry [ name = "cvc", purpose = cardCode, numeralWidth = tabular,
                    placeholder = { classroot.brand == "amex" ? "4 digits" : "3 digits" },
                    required = "Enter the security code from your card.",
                    value <-> app.cvc,
                    admit(t: string) -> string { return digitsOf(t).slice(0, 4) },
                    validate(v: string) -> string {
                        const n = codeLength(classroot.brand)
                        return digitsOf(v).length == n ? "" : "Enter the " + n + "-digit code " + (n == 4 ? "on the front of your card." : "on the back of your card.")
                        }
                    ]
                ]
            ]
        ],

    billing: FormSection [ visible = { !classroot.saved }, title = "Billing address",
        Field [ width = 100%, label = "Street address", stacked = true,
            Entry [ name = "line1", purpose = addressLine1, required = "Enter the street address for this card.",
                value <-> app.line1 ]
            ],
        Field [ width = 100%, label = "Address line 2 (optional)", stacked = true,
            Entry [ name = "line2", purpose = addressLine2, value <-> app.line2 ]
            ],
        Field [ width = 100%, label = "City", stacked = true,
            Entry [ name = "city", purpose = city, required = "Enter the town or city.",
                value <-> app.city ]
            ],
        View [ width = 100%,
            layout: SimpleLayout [ axis = x, spacing = 12 ],
            Field [ flexes = true, label = "Postal code", stacked = true,
                Entry [ name = "postalCode", purpose = postalCode, required = "Enter the postal code for this card.",
                    value <-> app.postalCode,
                    validate(v: string) -> string { return postalProblem(v, app.countryCode) }
                    ]
                ],
            Field [ flexes = true, label = "Country", stacked = true,
                Combobox [ width = 100%, name = "country", purpose = country, required = "Choose the country.",
                    items = { app.countries }, value <-> app.country ]
                ]
            ]
        ]
    ]

// ── The app ─────────────────────────────────────────────────────────────────

App [ appName = "The Venue", minWidth = 320,
    theme = { { ...SanFrancisco, ...Venue } },
    fill = { provided("theme").background },
    textColor = { provided("theme").text },

    // ── type: a book serif for the titles, the system's sans for the rest

    display: array = { ["Iowan Old Style", "Palatino Linotype", "Palatino", "Georgia", "serif"] },
    sans: array = { ["-apple-system", "system-ui", "Segoe UI", "Helvetica Neue", "sans-serif"] },
    fontFamily = { app.sans },

    // ── sizes

    wide: boolean = { app.width >= 820 },
    edge: number = { app.wide ? 0 : 20 },
    column: number = { Math.min(760, app.width - (app.wide ? 64 : 0)) },
    pitch: number = { app.wide ? Math.max(14, Math.min(30,
        (app.width - 340 - 48) / (app.hall.laid.value?.w ?? 34),
        (app.height - 60 - 40) / (app.hall.laid.value?.h ?? 34))) : 44 },
    deskLeft: number = { Math.max(32, (app.width - 960) / 2) },

    clock: Time [ tick = day ],

    // ── the service

    api: string = "http://127.0.0.1:8461",

    production: DataSource [ url = { app.api + "/api/production" }, schema = Production, credentials = include, autoFetch = true ],
    title: string = { app.production.value?.title ?? "The Marriage of Figaro" },
    hallName: string = { app.production.value?.hall ?? "Grand Hall" },
    performances: Performance[] = { app.production.value?.performances ?? [] },

    hall: Hall [ url = { app.perfId == "" ? "" : app.api + "/api/performances/" + app.perfId + "/seats" },
        onLoad() { app.centreHall() }
        ],

    // other people buying, while this person looks

    feed: EventStream [ url = { app.perfId == "" ? "" : app.api + "/api/events?performance=" + app.perfId },
        active = { app.perfId != "" && app.hall.loaded }, credentials = include, retry = 2,
        listenTo = ["seatchanged"],
        onMessage(e: StreamMessage) {
            const m = JSON.parse(e.data) as ({ seatId: string, status: string })
            if (m.status == "sold") app.seatTaken(m.seatId)
            }
        ],

    // ── who is signed in. The session is the service's (a cookie), so a
    //    reload asks it again; a 401 is simply "signed out".

    me: DataSource [ url = { app.api + "/api/me" }, schema = Profile, credentials = include, autoFetch = true ],
    signedIn: boolean = { app.me.loaded && !app.me.failed },
    profile: Profile? = { app.signedIn ? app.me.value : null },

    signEmail: string = "",
    signPassword: string = "",
    showPassword: boolean = false,
    signingIn: boolean = false,
    session: DataSource [ url = { app.api + "/api/session" }, method = "POST", credentials = include,
        body = { { email: app.signEmail.trim(), password: app.signPassword } },
        onLoad() {
            app.signPassword = ""
            app.signingIn = false
            app.me.fetch()
            }
        ],
    signOut: DataSource [ url = { app.api + "/api/session" }, method = "DELETE", credentials = include, format = "text",
        onLoad() { app.me.fetch(); app.toast.say("You're signed out.") }
        ],

    // ── where the person is: the performance is the address, the step past
    //    the hall is Back's to undo.

    location = "",
    waypoint = "",
    perfId: string = { /^p\d+$/.test(app.location) ? app.location : "" },
    perf: Performance? = { app.performances.find((p) => p.id == app.perfId) ?? null },
    receipt: Receipt? = { app.order.loaded ? app.order.value : null },
    atTickets: boolean = { app.waypoint == "tickets" && app.receipt != null },
    atCheckout: boolean = { app.waypoint == "checkout" && app.chosen.length > 0 },
    inHall: "hall" | "checkout" | "tickets" = { app.atTickets ? "tickets" : app.atCheckout ? "checkout" : "hall" },
    stage: "list" | "hall" | "checkout" | "tickets" = { app.perfId == "" ? "list" : app.inHall },

    chosen: string[] = [],
    notice: string = "",          // what the hall has to say: seats lost at payment

    trackChanges = ["perfId", "stage"],
    onChange(e: ChangeEvent) {
        if (e.changed.some((c) => c.name == "perfId")) { app.chosen = []; app.notice = ""; app.centred = "" }
        if (app.stage == "list") app.production.fetch()
        app.scrollTo(0)
        afterSettle(app.focusHeading)
        },

    // A new screen takes the focus to its heading, so a screen reader says
    // where the person now is and the next Tab goes on from there.

    focusHeading() {
        if (app.stage == "list") Focus.focus(app.list.intro.heading)
        else if (app.stage == "hall") Focus.focus(app.hallScreen.side.top.info.titleRow.heading)
        else if (app.stage == "checkout" && app.checkout != null) Focus.focus(app.checkout.heading)
        },

    openPerformance(id: string) { app.location = id; app.waypoint = "" },

    // The total answers a choice: it swells a little and settles.

    bump: number = 0,
    Spring [ attribute = bump, to = 0, stiffness = 380, damping = 16 ],

    toggleSeat(id: string) {
        if (!app.accessibilitySettings.reduceMotion) app.bump = 1
        if (app.chosen.includes(id)) app.chosen = app.chosen.filter((s) => s != id)
        else if ((app.hall.placed(id)?.seat.status ?? "sold") == "available") app.chosen = [...app.chosen, id]
        app.notice = ""
        },
    seatTaken(id: string) {
        app.hall.markSold(id)
        if (!app.chosen.includes(id)) return
        app.chosen = app.chosen.filter((s) => s != id)
        app.notice = (app.hall.placed(id)?.label ?? id) + " has just been taken by someone else, so it has left your choice."
        },
    seatsLost(ids: string[], message: string) {
        app.chosen = app.chosen.filter((s) => !ids.includes(s))
        app.notice = message
        app.waypoint = ""
        app.hall.fetch()
        },

    // On a phone the hall is larger than the screen; it opens on the middle
    // of the house, the stage at the top.

    centred: string = "",
    centreHall() {
        if (app.centred == app.perfId) return
        app.centred = app.perfId
        afterSettle(() => {
            const pane = app.hallScreen.pane
            const laid = app.hall.laid.value
            const front = laid == null || laid.floors.length == 0 ? 0 : (laid.cy + laid.floors[0].r0 - 2.6) * app.pitch
            pane.scrollTo(({ x: Math.max(0, (pane.map.width - pane.width) / 2), y: app.wide ? 0 : front }))
            })
        },

    // ── the choice, priced

    basketLines() -> Placed[] {
        return app.chosen.map((id) => app.hall.placed(id)).filter((p) => p != null)
            .sort((a, b) => a.tier - b.tier || a.row.localeCompare(b.row) || a.number - b.number) as Placed[]
        },
    basket: Dataset [ contents = { { lines: app.basketLines() } } ],
    subtotalCents: number = { app.basketLines().reduce((n, p) => n + p.priceCents, 0) },
    feesCents: number = { app.basketLines().reduce((n, p) => n + p.feeCents, 0) },
    totalCents: number = { app.subtotalCents + app.feesCents },
    summary: string = { seatCount(app.chosen.length) + " · " + money(app.subtotalCents) },
    seatList: string = { app.basketLines().map((p) => p.section + " " + p.row + p.number).join(", ") },

    // ── what is typed at checkout. Those that start from the account follow
    //    it until the person types; then they are theirs.

    email: string = { app.profile?.email ?? "" },
    cardName: string = { app.profile?.name ?? "" },
    cardNumber: string = "",
    expiry: string = "",
    cvc: string = "",
    line1: string = { app.profile?.billing.line1 ?? "" },
    line2: string = { app.profile?.billing.line2 ?? "" },
    city: string = { app.profile?.billing.city ?? "" },
    postalCode: string = { app.profile?.billing.postalCode ?? "" },
    countries: array = { [({ id: "US", label: "United States" }), ({ id: "CA", label: "Canada" }), ({ id: "GB", label: "United Kingdom" })] },
    country: object = { app.countries.find((c) => c.id == (app.profile?.billing.country ?? "US")) ?? null },
    countryCode: string = { (app.country as ({ id: string }) | null)?.id ?? "" },

    usableCards: SavedCardRec[] = { (app.profile?.cards ?? []).filter((c) => !isExpired(c.expMonth, c.expYear, app.clock.year, app.clock.month)) },
    payWith: string = { app.usableCards.length > 0 ? app.usableCards[0].id : "new" },
    savedCvc: string = "",
    savedBrand: string = { (app.profile?.cards ?? []).find((c) => c.id == app.payWith)?.brand ?? "" },
    paying: boolean = { app.payWith != "new" && app.signedIn },

    orderBody() -> object {
