// ── Formatting ──────────────────────────────────────────────────────────────
//
// Functions of their arguments: money, dates and times as the brief writes
// them, and what a card number says about itself.

script {
    const DAYS = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"]
    const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"]
    const USD = new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" })

    function money(cents: number): string { return USD.format(cents / 100) }

    // "2026-11-13T19:30" is the hall's own clock, as printed on a ticket, so
    // it is read as written, never through a time zone.
    function parts(iso: string): number[] {
        const m = /^(\d+)-(\d+)-(\d+)T(\d+):(\d+)/.exec(iso)
        return m == null ? [2000, 1, 1, 0, 0] : m.slice(1).map(Number)
    }
    function dayLabel(iso: string): string {
        const p = parts(iso)
        return DAYS[new Date(p[0], p[1] - 1, p[2]).getDay()] + " " + p[2] + " " + MONTHS[p[1] - 1]
    }
    function weekday(iso: string): string { const p = parts(iso); return DAYS[new Date(p[0], p[1] - 1, p[2]).getDay()] }
    function dayOfMonth(iso: string): string { return "" + parts(iso)[2] }
    function monthLabel(iso: string): string { return MONTHS[parts(iso)[1] - 1] }
    function timeLabel(iso: string): string {
        const p = parts(iso)
        const h = p[3] % 12 == 0 ? 12 : p[3] % 12
        return h + ":" + (p[4] < 10 ? "0" : "") + p[4] + (p[3] < 12 ? "am" : "pm")
    }
    function seatCount(n: number): string { return n + (n == 1 ? " seat" : " seats") }

    // ── cards

    function digitsOf(s: string): string { return s.replace(/\D/g, "") }
    function cardBrand(d: string): string {
        if (/^4/.test(d)) return "visa"
        if (/^3[47]/.test(d)) return "amex"
        if (/^(5[1-5]|2(2[2-9]|[3-6]\d|7[01]|720))/.test(d)) return "mastercard"
        if (/^(6011|65|64[4-9])/.test(d)) return "discover"
        return ""
    }
    function brandName(b: string): string {
        return b == "visa" ? "Visa" : b == "mastercard" ? "Mastercard" : b == "amex" ? "American Express" : b == "discover" ? "Discover" : ""
    }
    function cardLength(b: string): number { return b == "amex" ? 15 : 16 }
    function codeLength(b: string): number { return b == "amex" ? 4 : 3 }

    // grouped as printed: 4-6-5 on an American Express card, fours on the rest
    function groupCard(d: string): string {
        const groups = cardBrand(d) == "amex" ? [4, 6, 5] : [4, 4, 4, 4]
        const out: string[] = []
        let at = 0
        for (const g of groups) { if (at >= d.length) break; out.push(d.slice(at, at + g)); at = at + g }
        return out.join(" ")
    }
    function luhn(d: string): boolean {
        let sum = 0
        for (let i = 0; i < d.length; i++) {
            let n = Number(d[d.length - 1 - i])
            if (i % 2 == 1) { n = n * 2; if (n > 9) n = n - 9 }
            sum = sum + n
        }
        return d.length > 0 && sum % 10 == 0
    }
    function cardProblem(text: string): string {
        const d = digitsOf(text)
        const b = cardBrand(d)
        if (b == "") return "That isn't a card number we recognise. Check the first digits."
        if (d.length < cardLength(b)) return "The card number is too short."
        if (!luhn(d)) return "That card number isn't valid. Check it against your card."
        return ""
    }

    // "1228" or "12/28" → "12 / 28"; a leading 2–9 is a month on its own
    function tidyExpiry(t: string): string {
        let d = digitsOf(t).slice(0, 4)
        if (d.length == 1 && d > "1") d = "0" + d
        return d.length <= 2 ? d : d.slice(0, 2) + " / " + d.slice(2)
    }
    function expiryParts(t: string): number[] {
        const d = digitsOf(t)
        return d.length == 4 ? [Number(d.slice(0, 2)), 2000 + Number(d.slice(2))] : [0, 0]
    }
    function expiryProblem(t: string, year: number, month: number): string {
        const d = digitsOf(t)
        if (d.length < 4) return "Enter the expiry as MM / YY."
        const p = expiryParts(t)
        if (p[0] < 1 || p[0] > 12) return "There's no month " + d.slice(0, 2) + ". Use 01 to 12."
        if (p[1] < year || (p[1] == year && p[0] < month)) return "That card has expired."
        return ""
    }
    function isExpired(m: number, y: number, year: number, month: number): boolean {
        return y < year || (y == year && m < month)
    }

    // ── addresses

    function postalProblem(code: string, country: string): string {
        const c = code.trim()
        if (country == "US") return /^\d{5}(-\d{4})?$/.test(c) ? "" : "Enter a 5-digit ZIP code, like 94103."
        if (country == "CA") return /^[A-Za-z]\d[A-Za-z] ?\d[A-Za-z]\d$/.test(c) ? "" : "Enter a postal code like K1A 0B1."
        if (country == "GB") return /^[A-Za-z]{1,2}\d[A-Za-z\d]? ?\d[A-Za-z]{2}$/.test(c) ? "" : "Enter a postcode like SW1A 1AA."
        return ""
    }

    // A barcode's bars, from its code: widths 1–3, bar and space in turn.
    function bars(code: string): number[] {
        const out: number[] = [2, 1, 1, 2]
        for (let i = 0; i < code.length; i++) {
            const c = code.charCodeAt(i) * 7 + i * 13
            out.push(1 + (c % 3), 1 + ((c >> 2) % 2), 1 + ((c >> 3) % 3), 1 + ((c >> 1) % 2))
        }
        out.push(2, 1, 1, 2)
        return out
    }
    }

class Chevron [ ink: Color = { provided("theme").textFaint },
    left: boolean = false,
    draw(d: DrawContext) {
        d.strokeStyle = ink
        d.lineWidth = 1.8
        d.lineCap = "round"
        d.lineJoin = "round"
        d.beginPath()
        const w = d.w, h = d.h
        if (left) { d.moveTo(w * 0.75, h * 0.15); d.lineTo(w * 0.25, h * 0.5); d.lineTo(w * 0.75, h * 0.85) }
        else { d.moveTo(w * 0.25, h * 0.15); d.lineTo(w * 0.75, h * 0.5); d.lineTo(w * 0.25, h * 0.85) }
        d.stroke()
        }
    ]


// ── A seat ──────────────────────────────────────────────────────────────────
//
// A seat is a toggle: free, yours, or gone, told apart by shape as well as
// colour — a free seat is a solid disc in its tier's ink, yours is larger, in
// the accent, with a check, and a seat that is gone is a small hollow ring.
// The view is the whole pitch, so the target is as large as the seat's share
// of the floor; the disc inside it is what shows.

class Seat extends Control [
    pitch: number = { provided("pitch") },
    seatId: string = { :id ?? "" },
    gone: boolean = { :seat.status == "sold" },
    mine: boolean = { app.chosen.includes(classroot.seatId) },
    x = { (:ux ?? 0) * classroot.pitch - classroot.pitch / 2 },
    y = { (:uy ?? 0) * classroot.pitch - classroot.pitch / 2 },
    width = { classroot.pitch },
    height = { classroot.pitch },
    disabled = { classroot.gone },
    accessibilityRole = toggle,
    accessibilityLabel = { :label ?? "" },
    accessibilityPressed = { classroot.mine },
    accessibilityHint = { classroot.gone ? "Taken." : money(:priceCents ?? 0) + (classroot.mine ? ", in your choice." : ", free.") },
    tooltipLabel = { app.wide ? (:label ?? "") + " · " + (classroot.gone ? "taken" : money(:priceCents ?? 0)) : "" },
    press() { app.toggleSeat(classroot.seatId) },

    // Left and Right go along the row (the hall is one Tab stop, and its
    // arrows move among the seats); Up and Down go to the row in front or
    // behind, to the free seat nearest across.

    onKeyDown(e: KeyEvent) {
        if (e.key != "ArrowUp" && e.key != "ArrowDown") { super.onKeyDown(e); return }
        const id = app.hall.neighbour(classroot.seatId, e.key == "ArrowDown" ? 1 : -1)
        const to = (parent.childViews as Seat[]).find((v) => v.seatId == id)
        if (to != null) Focus.focus(to)
        },

    // The disc's size, on a spring: choosing a seat makes it swell a little
    // past its size and settle, so the commitment registers in the seat.

    size: number = 0.66,
    want: number = { classroot.mine ? 0.86 : classroot.gone ? 0.4 : 0.66 },
    Spring [ attribute = size, to = { classroot.want }, stiffness = 520, damping = 16 ],
    lift: number = { classroot.hovered || classroot.focused ? 1.12 : 1 },

    disc: View [ x = center, y = center,
        width = { classroot.pitch * classroot.size * classroot.lift }, height = { this.width },
        cornerRadius = { this.width / 2 },
        fill = { provided("theme").tierInks[:tier ?? 0] },
        State [ applied = { classroot.mine }, fill = { provided("theme").accent } ],
        State [ applied = { classroot.gone }, fill = null, stroke = { stroke(1.5, provided("theme").gone) } ],
        number: Text [ x = center, y = center, visible = { classroot.pitch >= 34 && !classroot.mine && !classroot.gone },
            fontSize = 11, fontWeight = semibold, numeralWidth = tabular, textColor = #FFFFFF,
            accessibilityHidden = true, text = { "" + (:number ?? "") } ],
        Check [ x = center, y = center, visible = { classroot.mine },
            width = { parent.width * 0.5 }, height = { this.width } ]
        ]
    ]


class Check [ ink: Color = #FFFFFF,
    draw(d: DrawContext) {
        d.strokeStyle = ink
        d.lineWidth = Math.max(1.6, d.w * 0.16)
        d.lineCap = "round"
        d.lineJoin = "round"
        d.beginPath()
        d.moveTo(d.w * 0.12, d.h * 0.52)
        d.lineTo(d.w * 0.4, d.h * 0.8)
        d.lineTo(d.w * 0.9, d.h * 0.2)
        d.stroke()
        }
    ]


// ── The hall's floor ────────────────────────────────────────────────────────
//
// Drawn beneath the seats: the stage everything faces, and a floor for each
// tier, curved as its rows are. A scene, so it is drawn rather than laid out.

class HallFloor [
    pitch: number = { provided("pitch") },
    accessibilityHidden = true,
    draw(d: DrawContext) {
        const laid = app.hall.laid.value
        if (laid == null) return
        const p = pitch
        const cx = laid.cx * p, cy = laid.cy * p
        const t = provided("theme")
        for (const f of laid.floors) {
            const a = f.half
            d.beginPath()
            d.arc(cx, cy, f.r1 * p, Math.PI / 2 - a, Math.PI / 2 + a)
            d.arc(cx, cy, f.r0 * p, Math.PI / 2 + a, Math.PI / 2 - a, true)
            d.closePath()
            d.fillStyle = t.floors[f.tier]
            d.fill()
        }
        // the stage: a curved apron, the darkest thing in the room
        const s0 = laid.floors.length > 0 ? laid.floors[0].r0 : 20
        const a = 0.34
        d.beginPath()
        d.arc(cx, cy, (s0 - 1.5) * p, Math.PI / 2 - a, Math.PI / 2 + a)
        d.arc(cx, cy, (s0 - 4.2) * p, Math.PI / 2 + a, Math.PI / 2 - a, true)
        d.closePath()
        d.fillStyle = t.stage
        d.fill()
        }
    ]

// ── The overview ────────────────────────────────────────────────────────────
//
// On a phone the hall is shown at a finger's size, larger than the screen, so
// a small drawing of the whole house sits in the corner: every seat a dot, and
// a frame round what the pane shows. A press on it moves the pane there.

class Overview extends Control [ cornerRadius = 12, clip = true,
    fill = { provided("theme").surface },
    stroke = { stroke(1, provided("theme").line) },
    shadow = { shadow(0, 6, 18, 0x1E1B1626) },
    pane: View? = null,
    accessibilityLabel = "Overview of the hall",
    accessibilityHint = "Moves the view of the hall to the part you press.",
    scaleBy: number = { (this.width - 12) / Math.max(1, app.hall.laid.value?.w ?? 1) },
    onClick(e: PointerEvent) {
        const laid = app.hall.laid.value
        if (laid == null || pane == null) return
        const p = provided("pitch")
        const ux = (e.x - 6) / scaleBy, uy = (e.y - 6) / scaleBy
        pane.scrollTo(({ x: ux * p - pane.width / 2, y: uy * p - pane.height / 2 }), ({ duration: 300 }))
        },
    draw(d: DrawContext) {
        const laid = app.hall.laid.value
        if (laid == null) return
        const t = provided("theme")
        const k = scaleBy
        const chosen = app.chosen
        for (const s of laid.seats) {
            const mine = chosen.includes(s.id)
            d.beginPath()
            d.arc(6 + s.ux * k, 6 + s.uy * k, mine ? k * 0.55 : k * 0.36, 0, 6.3)
            d.fillStyle = mine ? t.accent : s.seat.status == "sold" ? t.line : t.tierInks[s.tier]
            d.fill()
        }
        if (pane != null) {
            const p = provided("pitch")
            d.strokeStyle = t.text
            d.lineWidth = 1.5
            d.strokeRect(6 + pane.scrollX / p * k, 6 + pane.scrollY / p * k, pane.width / p * k, pane.height / p * k)
        }
        }
    ]

class Barcode [
    code: string = "",
    ink: Color = { provided("theme").text },
    draw(d: DrawContext) {
        const b = bars(code)
        const units = b.reduce((s, n) => s + n, 0)
        const u = d.w / units
        let x = 0
        d.fillStyle = ink
        b.forEach((n, i) => {
            if (i % 2 == 0) d.fillRect(x, 0, n * u, d.h)
            x = x + n * u
            })
        }
    ]
