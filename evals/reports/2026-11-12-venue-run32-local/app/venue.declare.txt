/*
# The Venue

The box office for *The Marriage of Figaro* in the Grand Hall: pick one of
twenty performances, choose seats in a drawn plan of the house while other
people buy around you, pay with a saved card or a new one, and come away with
tickets to hold up at the door. One program for a phone and a desk.

How to read it:

- **The records** come from the service at `app.api`: the production, one
  performance's seat map (`SeatMap`, which also lays the house out), the
  signed-in account, and the order. Each is a `DataSource`; nothing copies them.
- **Where the person is** sits on the App: the performance (in the URL, so it
  can be shared), the step (`waypoint`: Back walks it, a stranger never sees
  it), the seats they hold, and what they have typed at checkout.
- **Everything on screen derives from those.** Seats taken elsewhere arrive on
  an event stream and are written into the seat map; the hall, the totals and
  the checkout follow.

The look — the theme, the drawn seat, the barcode, the small controls — is in
`venue-parts.declare`.
*/

include [ "venue-parts.declare" ]

// ── The seat map, and the house it lays out ─────────────────────────────────
//
// The plan is a fan: every row is an arc about a point behind the stage, so
// each seat faces it. Sections follow front to back with a gap for their
// name; long rows split into three blocks with an aisle either side.

class SeatMap extends DataSource [ autoFetch = true, schema = SeatsReply,
    plan: Dataset [ schema = Plan, contents = { classroot.layOut() } ],

    sections() -> SectionRec[] { return this.value?.sections ?? [] },
    countWhere(status: string) -> number {
        return this.sections().reduce((n, s) => n + s.rows.reduce((m, r) => m + r.seats.filter((x) => x.status == status).length, 0), 0)
        },
    free: number = { classroot.countWhere("available") },
    total: number = { classroot.countWhere("available") + classroot.countWhere("sold") },

    layOut() -> Plan {
        const R0 = 15
        const aisle = 0.9
        const top = 2.6
        let depth = 2.6
        const seats: PlanSeat[] = []
        const labels: PlanLabel[] = []
        this.sections().forEach((s, si) => {
            if (si > 0) depth = depth + 1.8
            labels.push(({ id: s.key, name: s.name, priceCents: s.priceCents, cx: 0, cy: depth - 1.2 }))
            s.rows.forEach((r) => {
                const n = r.seats.length
                const side = n >= 20 ? Math.round(n * 0.22) : 0
                const span = n - 1 + (side > 0 ? 2 * aisle : 0)
                const R = R0 + depth
                r.seats.forEach((seat, i) => {
                    const u = i + (side > 0 && i >= side ? aisle : 0) + (side > 0 && i >= n - side ? aisle : 0)
                    const along = span / 2 - u       // seat 1 is at the stage's left: the audience's right
                    const th = along / R
                    seats.push(({ id: seat.id, seat: seat, section: s.name, row: r.name, tier: si,
                        priceCents: s.priceCents, feeCents: s.feeCents,
                        cx: R * Math.sin(th), cy: R * Math.cos(th) - R0, turn: -th * 180 / Math.PI }))
                    })
                depth = depth + 1.12
                })
            })
        const half = seats.reduce((m, p) => Math.max(m, Math.abs(p.cx)), 8) + 1.2
        return ({
            seats: seats.map((p) => ({ ...p, cx: p.cx + half, cy: p.cy + top })),
            labels: labels.map((l) => ({ ...l, cx: half, cy: l.cy + top })),
            width: half * 2,
            height: depth + top + 0.4
            })
        },

    seatAt(id: string) -> PlanSeat? { return (this.plan.value?.seats ?? []).find((p) => p.id == id) ?? null },
    isFree(id: string) -> boolean { return this.seatAt(id)?.seat.status == "available" },

    markSold(id: string) {
        const secs = this.sections()
        for (let si = 0; si < secs.length; si++) {
            const rows = secs[si].rows
            for (let ri = 0; ri < rows.length; ri++) {
                const i = rows[ri].seats.findIndex((x) => x.id == id)
                if (i >= 0) {
                    if (rows[ri].seats[i].status != "sold") this.set(["sections", si, "rows", ri, "seats", i, "status"], "sold")
                    return
                }
            }
        }
        }
    ]

// The stage every row faces: a dark apron curving toward the house.

class Stage [ width = 100%, height = 100%, accessibilityHidden = true,
    draw(d: DrawContext) {
        const p = app.pitch
        const t = provided("theme")
        const cx = this.width / 2
        const cy = (2.6 - 15) * p
        const ra = 15.7 * p
        const a = Math.asin(7.5 / 15.7)
        const back = 0.5 * p
        d.fillStyle = t.stage
        d.beginPath()
        d.moveTo(cx - ra * Math.sin(a), back)
        d.lineTo(cx + ra * Math.sin(a), back)
        d.lineTo(cx + ra * Math.sin(a), cy + ra * Math.cos(a))
        d.arc(cx, cy, ra, Math.PI / 2 - a, Math.PI / 2 + a)
        d.closePath()
        d.fill()
        },
    TextLabel [ x = center, y = { 0.95 * app.pitch }, fontSize = { Math.max(11, app.pitch * 0.45) }, fontWeight = semibold,
        letterSpacing = 4, textColor = { provided("theme").stageText }, text = "STAGE" ]
    ]


// The whole house at a glance, for a phone: every seat a dot, the part the
// pane shows outlined. Press or drag on it to look somewhere else.

class HallMap [ height = 96, accessibilityHidden = true,
    planW: number = { app.seatMap.plan.value?.width ?? 36 },
    planH: number = { app.seatMap.plan.value?.height ?? 30 },
    k: number = { Math.min(this.width / classroot.planW, this.height / classroot.planH) },
    ox: number = { (this.width - classroot.planW * classroot.k) / 2 },
    look(px: number, py: number) { app.lookAt((px - ox) / (planW * k), py / (planH * k)) },
    onClick(e: PointerEvent) { look(e.x, e.y) },
    onDrag(e: PointerEvent) { look(e.x, e.y) },
    draw(d: DrawContext) {
        const t = provided("theme")
        const seats = app.seatMap.plan.value?.seats ?? []
        const chosen = app.chosen
        const r = Math.max(1.2, k * 0.36)
        seats.forEach((s) => {
            const mine = chosen.includes(s.id)
            d.fillStyle = mine ? t.accent : s.seat.status == "sold" ? t.soldInk : t.seatInk
            d.beginPath()
            d.arc(ox + s.cx * k, s.cy * k, mine ? r * 1.6 : r, 0, Math.PI * 2)
            d.fill()
            })
        d.fillStyle = t.stage
        d.fillRect(ox + (planW / 2 - 6) * k, 0.4 * k, 12 * k, 1.6 * k)
        const pane = app.hall.pane
        const sw = Math.max(1, pane.scene.width)
        const sh = Math.max(1, pane.scene.height)
        const vx = pane.scrollX / sw
        const vy = pane.scrollY / sh
        const vw = Math.min(1, pane.width / sw)
        const vh = Math.min(1, pane.height / sh)
        d.strokeStyle = t.text
        d.lineWidth = 1.5
        d.strokeRect(ox + vx * planW * k, vy * planH * k, vw * planW * k, vh * planH * k)
        }
    ]
