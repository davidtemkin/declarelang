// No presentations yet: a small drawn scene of slides being made, and the one
// thing to do.

class EmptyHome [ padding = [40, 0, 40, 0],
    layout: SimpleLayout [ axis = y, spacing = 14, align = center ],
    art: View [ width = 300, height = 180,
        draw(d: DrawContext) {
            // three slides fanned out, the front one with a shape and a title on it
            const card = (x: number, y: number, rot: number, fill: string) => {
                d.save(); d.translate(x, y); d.rotate(rot)
                d.shadowColor = "#1A183026"; d.shadowBlur = 18; d.shadowOffsetY = 8
                d.fillStyle = fill; d.beginPath(); d.roundRect(-90, -51, 180, 102, 8); d.fill()
                d.restore()
                }
            card(105, 96, -0.16, "#FFC23D")
            card(195, 92, 0.14, "#FF6A4D")
            card(150, 86, 0, "#FFFFFF")
            d.fillStyle = "#6A4DFF"; d.beginPath(); d.roundRect(72, 50, 70, 46, 10); d.fill()
            d.fillStyle = "#1FBF91"; d.beginPath(); d.arc(186, 72, 22, 0, Math.PI * 2); d.fill()
            d.fillStyle = "#1A1830"; d.beginPath(); d.roundRect(72, 108, 120, 9, 4.5); d.fill()
            d.fillStyle = "#D3CEC4"; d.beginPath(); d.roundRect(72, 123, 80, 7, 3.5); d.fill()
            d.fillStyle = "#6A4DFF"; d.beginPath(); d.moveTo(232, 132); d.lineTo(252, 144); d.lineTo(232, 156); d.closePath(); d.fill()
            }
        ],
    Text [ fontFamily = { app.brandFace }, fontWeight = 800, fontSize = 26, letterSpacing = -0.6, text = "Make your first presentation" ],
    Text [ width = 420, textAlign = center, fontSize = 15, textColor = { provided("theme").textMuted },
        text = "Slides, shapes and words, saved as you go. Start with a blank slide; it takes about a minute to make three." ],
    Button [ label = "New presentation", primary = true, iconLeft = "PlusGlyph", height = 40,
        onClick() { app.newPresentation() }
        ]
    ]


// ── Presenting ──────────────────────────────────────────────────────────────
//
// One slide at a time, as large as the screen allows, on black. A click, the
// arrow keys, space and Page Up/Down move through; Escape ends, back on the
// slide the person ended on.

class Presenter [ fill = black, cursor = { classroot.idle ? "none" : "default" },
    at: number = 0,
    total: number = { app.doc.slides().length },
    s: number = { Math.min(this.width / 1280, this.height / 720) },
    idle: boolean = false,
    nudge: number = 0,
    accessibilityLabel = "Presentation", accessibilityRole = region,
    onClick() { app.presentStep(1) },
    onPointerMove() { this.wake() },
    wake() {
        idle = false
        nudge = nudge + 1
        const n = nudge
        afterDelay(1800, () => { if (n == this.nudge) this.idle = true })
        },
    onInit() { this.wake() },
    slide: SlideView [ x = { (parent.width - 1280 * parent.s) / 2 }, y = { (parent.height - 720 * parent.s) / 2 },
        scale = { parent.s }, accessibilityRole = group, accessibilityLabel = { "Slide " + (classroot.at + 1) + " of " + classroot.total },
        datapath = { app.doc.slides()[classroot.at] ?? null } ],
    hint: View [ x = center, y = { parent.height - 56 }, height = 32, padding = [0, 14, 0, 14], cornerRadius = 16,
        fill = 0x1A1830CC,
        Spring [ attribute = opacity, to = { classroot.idle ? 0 : 1 }, stiffness = 120, damping = 20 ],
        layout: SimpleLayout [ axis = x, spacing = 14, align = center ],
        TextLabel [ fontSize = 12.5, fontWeight = 600, textColor = white, numeralWidth = tabular,
            text = { (classroot.at + 1) + " / " + classroot.total } ],
        TextLabel [ fontSize = 12.5, textColor = 0xFFFFFFB3, text = "← → to move  ·  Esc to end" ]
        ]
    ]
