the tools, undo, the save state and Present), the slide list down the left
side, and the inspector down the right. The canvas itself is in
`canvas.declare`; what every command does is a method on the App.

// ── The slide list ──────────────────────────────────────────────────────────
//
// The slides in order, each a small live picture of itself. Reordering is a
// drag: while it lasts the list holds still and a gap-line shows where the
// slide will land; on release the slide moves in the data and the list's
// layout glides every thumbnail to its new place.

class RailLayout extends TweenLayout [ duration = 260,
    pitch: number = 112,
    place() {
        const w = this.contentExtent("width")
        return this.laid().map((c, i) => ({ x: 0, y: i * this.pitch, w: w, h: this.pitch - 8 }))
        }
    ]

class SlideThumb extends Control [ width = 100%,
    slideKey: string = { :key ?? "" },
    n: number = { rowIndex + 1 },
    here: boolean = { app.currentKey == classroot.slideKey },
    dragged: boolean = { app.rail.list.dragKey == classroot.slideKey },
    conflicted: boolean = { app.sync.conflictOf(classroot.slideKey) != null },
    accessibilityLabel = { "Slide " + classroot.n },
    tooltipLabel = { "Slide " + classroot.n },
    accessibilityRole = button,
    accessibilityPressed = { classroot.here },
    press() { app.goTo(classroot.slideKey); app.focusArea = "rail" },
    onKeyDown(e: KeyEvent) {
        if (e.key == "ArrowDown" || e.key == "ArrowRight") app.step(1, true)
        else if (e.key == "ArrowUp" || e.key == "ArrowLeft") app.step(-1, true)
        else super.onKeyDown(e)
        },
    onContextMenu(e: PointerEvent) { app.goTo(classroot.slideKey); app.slideMenu.open(this, e) },

    // dragging to reorder: a few points of slack, so a click is a click

    _y0: number = 0,
    onPointerDown(e: PointerEvent) {
        super.onPointerDown()
        _y0 = e.y
        },
    onPointerMove(e: PointerEvent) {
        const list = app.rail.list
        if (list.dragKey == "" && Math.abs(e.deltaY) < 5) return
        if (list.dragKey == "") { list.dragKey = classroot.slideKey; list.grabDy = _y0 - 4 }
        const top = list.rootOrigin().y
        list.dragY = e.rootY - top + list.scrollY
        const n = app.doc.slides().length
        list.slot = Math.max(0, Math.min(n, Math.round((list.dragY - list.grabDy) / list.pitch)))
        },
    onPointerUp(e: PointerUpEvent) {
        const list = app.rail.list
        if (list.dragKey == "") return
        const slot = list.slot
        list.dragKey = ""
        list.slot = -1
        if (!e.canceled) app.moveSlideTo(classroot.slideKey, slot)
        },

    trackChanges = ["here"],
    onChange() { if (this.here) this.scrollIntoView() },

    opacity = { classroot.dragged ? 0.35 : 1 },
    num: Text [ x = 4, y = 4, width = 20, textAlign = right, fontSize = 12, fontWeight = { classroot.here ? 700 : 500 },
        numeralWidth = tabular,
        textColor = { classroot.here ? provided("theme").accent : provided("theme").textFaint }, text = { "" + classroot.n } ],
    frame: View [ x = 30, y = 2, width = 160, height = 90, cornerRadius = 6, clip = true,
        shadow = { classroot.here ? shadow(0, 4, 12, 0x6A4DFF33) : shadow(0, 1, 3, 0x1A18301F) },
        SlideView [ scale = { 160 / 1280 } ]
        ],
    ring: View [ x = 27, y = -1, width = 166, height = 96, cornerRadius = 8,
        stroke = { classroot.here ? stroke(2.5, provided("theme").accent) : classroot.hovered ? stroke(1.5, provided("theme").lineStrong) : stroke(1, provided("theme").line) } ],
    flag: View [ visible = { classroot.conflicted }, x = 166, y = 8, width = 20, height = 20, cornerRadius = 10,
        fill = { provided("theme").sun }, tooltipLabel = "Changed in another window",
        Glyph [ x = center, y = center, iconSize = 12, shape = "warn", ink = { provided("theme").ink } ]
        ]
    ]

// A hex value, typed: what the well shows, as text, for a person who knows
// the colour they want. "none" (or an empty field) clears it, where allowed.

class HexField extends TextInput [ width = 96, height = 28, fontSize = 12.5, padding = [0, 8, 0, 8],
    numeralWidth = tabular,
    which: string = "",
    hex: string = "",
    mixed: boolean = false,
    noneAllowed: boolean = true,
    accessibilityLabel = { (classroot.which == "background" ? "Background" : classroot.which == "stroke" ? "Outline" : classroot.which == "text" ? "Text colour" : "Fill") + " hex value" },
    commitOn = "end",
    placeholder = { classroot.disabled ? "" : classroot.mixed ? "Mixed" : "None" },
    value = { classroot.disabled || classroot.mixed ? "" : classroot.hex },
    commit(v: string) {
        const t = v.trim()
        if (t == "" || t.toLowerCase() == "none") { if (noneAllowed && !mixed) app.applyColor(which, null); return }
        const h = normalizeHex(t)
        if (h != null) app.applyColor(which, h)
        }
    ]


// A number with its unit, typed or stepped with the arrow keys (shift for
// ten). Empty with "Mixed" when the selection's values differ.

class NumberField extends TextInput [ width = 72, height = 28, fontSize = 12.5, padding = [0, 8, 0, 8],
    numeralWidth = tabular, textAlign = right,
    label: string = "",
    unit: string = "",
    num: number = 0,
    mixed: boolean = false,
    min: number = 0,
    max: number = 1000,
    accessibilityLabel = { classroot.label },
    commitOn = "end",
    placeholder = { classroot.disabled ? "" : classroot.mixed ? "Mixed" : "" },
    value = { classroot.disabled || classroot.mixed ? "" : "" + Math.round(classroot.num * 10) / 10 },
    admit(t: string) -> string | null { return /^[0-9]*\.?[0-9]*$/.test(t) ? t : null },
    onKeyDown(e: KeyEvent) {
        if (e.key != "ArrowUp" && e.key != "ArrowDown") { super.onKeyDown(e); return }
        const step = (e.key == "ArrowUp" ? 1 : -1) * (e.shift ? 10 : 1)
        this.deliver(Math.max(min, Math.min(max, (mixed ? 0 : num) + step)))
        },
    deliver(v: number) { },
    commit(v: string) {
        if (v.trim() == "") return
        const n = Number(v)
        if (!isNaN(n)) this.deliver(Math.max(min, Math.min(max, n)))
        }
    ]
