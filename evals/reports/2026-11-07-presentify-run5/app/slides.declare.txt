/*
# Presentify: drawing a slide

One class draws a slide, at any size: `SlideView` is 1280 × 720 points and is
shown smaller or larger with `scale`, so the editor, the slide list, the
presentations grid and the presentation itself are the same views at
different scales, and look exactly alike.

It binds to a slide record (`:background`, `:elements`), from the open Doc or
straight from the service, and replicates one view per element, the class
chosen by the element's kind. A group's view replicates its children the
same way, so groups nest as deep as the data does.

Only the editor's own slide is `live`: there a text box can be edited in
place, and reports the height its text lays out to (the format keeps it).
*/

class SlideView [ width = 1280, height = 720, pivotX = 0, pivotY = 0, clip = true,
    live: boolean = false,
    fill = { app.paint(:background ?? "#FFFFFF") },
    ShapeView [ datapath = :elements[],
        classFor = { :kind == "group" ? GroupView : :kind == "text" ? TextView : :kind == "line" ? LineView : :kind == "ellipse" ? EllipseView : RectView } ]
    ]


// Every element's view. Its fields are read once, here, as named values.

class ShapeView [
    elId: string = { :id ?? "" },
    fillHex: string = { :fill ?? "" },
    lineHex: string = { :stroke?.color ?? "" },
    lineW: number = { :stroke?.width ?? 0 },
    edge: Stroke? = { classroot.lineHex == "" || classroot.lineW <= 0 ? null : stroke(classroot.lineW, app.paint(classroot.lineHex)) }
    ]


class RectView extends ShapeView [
    x = { :x ?? 0 }, y = { :y ?? 0 }, width = { :width ?? 0 }, height = { :height ?? 0 },
    fill = { classroot.fillHex == "" ? null : app.paint(classroot.fillHex) },
    stroke = { classroot.edge },
    cornerRadius = { :cornerRadius ?? 0 }
    ]


// An ellipse is drawn: a rounded box can only make a pill.

class EllipseView extends ShapeView [
    x = { :x ?? 0 }, y = { :y ?? 0 }, width = { :width ?? 0 }, height = { :height ?? 0 },
    draw(d: DrawContext) {
        const w = d.w, h = d.h
        if (w <= 0 || h <= 0) return
        if (fillHex != "") {
            d.fillStyle = fillHex
            d.beginPath(); d.ellipse(w / 2, h / 2, w / 2, h / 2, 0, 0, Math.PI * 2); d.fill()
        }
        if (lineHex != "" && lineW > 0) {
            const i = lineW / 2
            d.strokeStyle = lineHex
            d.lineWidth = lineW
            d.beginPath(); d.ellipse(w / 2, h / 2, Math.max(0.5, w / 2 - i), Math.max(0.5, h / 2 - i), 0, 0, Math.PI * 2); d.stroke()
        }
        }
    ]


// A line's view is the box around it, with room for the stroke's width.

class LineView extends ShapeView [
    ax: number = { :x1 ?? 0 },
    ay: number = { :y1 ?? 0 },
    bx: number = { :x2 ?? 0 },
    by: number = { :y2 ?? 0 },
    pad: number = { classroot.lineW / 2 + 2 },
    x = { Math.min(classroot.ax, classroot.bx) - classroot.pad },
    y = { Math.min(classroot.ay, classroot.by) - classroot.pad },
    width = { Math.abs(classroot.bx - classroot.ax) + 2 * classroot.pad },
    height = { Math.abs(classroot.by - classroot.ay) + 2 * classroot.pad },
    draw(d: DrawContext) {
        if (lineHex == "" || lineW <= 0) return
        const ox = Math.min(ax, bx) - pad, oy = Math.min(ay, by) - pad
        d.strokeStyle = lineHex
        d.lineWidth = lineW
        d.lineCap = "round"
        d.beginPath(); d.moveTo(ax - ox, ay - oy); d.lineTo(bx - ox, by - oy); d.stroke()
        }
    ]


// A text box: one block, one style, wrapping within its width and as tall
// as its lines. Being edited, the same block is a native text field laid
// exactly over its own words, so the caret sits where the letters are; the
// words underneath stay (invisible) and keep the box its height as it grows.

class TextView extends ShapeView [
    x = { :x ?? 0 }, y = { :y ?? 0 }, width = { :width ?? 0 },
    content: string = { :text?.content ?? "" },
    font: array = { [:text?.font ?? "Helvetica Neue", "Helvetica Neue", "Helvetica", "Arial", "sans-serif"] },
    size: number = { :text?.size ?? 32 },
    weight: number = { :text?.weight ?? 400 },
    slant: boolean = { :text?.italic ?? false },
    ink: Color = { app.paint(:text?.color ?? "#1A1830") },
    alignTo: "left" | "center" | "right" = { (:text?.align ?? "left") as "left" | "center" | "right" },
    editing: boolean = { provided("live") && app.editingId == classroot.elId },
    fontFamily = { classroot.font }, fontSize = { classroot.size }, fontWeight = { classroot.weight },
    textColor = { classroot.ink },

    words: Text [ width = 100%, italic = { classroot.slant }, textAlign = { classroot.alignTo },
        opacity = { classroot.editing ? 0 : 1 },
        text = { classroot.content == "" ? " " : classroot.content } ],

    field: TextInput [ exists = { classroot.editing }, x = 0, y = 0, width = 100%,
        height = { classroot.words.height }, padding = 0, fill = null, stroke = null, cornerRadius = 0,
        multiline = true, wrap = true, enterSubmits = false,
        italic = { classroot.slant }, textAlign = { classroot.alignTo },
        accessibilityLabel = "Text box",
        value = { classroot.content },
        commit(v: string) { app.typed(classroot.elId, v) },
        onKeyDown(e: KeyEvent) { if (e.key == "Escape") app.endTextEdit(); else super.onKeyDown(e) },
        onInit() { afterSettle(() => Focus.focus(this)) }
        ],

    // The format keeps a text box's height as last laid out.

    trackChanges = ["height"],
    onChange() { if (provided("live")) app.laidOut(classroot.elId, this.height) }
    ]


// A group has no box of its own: its children keep their slide positions.

class GroupView extends ShapeView [
    ShapeView [ datapath = :children[],
        classFor = { :kind == "group" ? GroupView : :kind == "text" ? TextView : :kind == "line" ? LineView : :kind == "ellipse" ? EllipseView : RectView } ]
    ]
