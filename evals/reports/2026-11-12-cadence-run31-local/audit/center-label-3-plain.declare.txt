// What a content-sized pill needs: no x = center at all. Its width is
// label + padding, so the symmetric padding centers the label.
class Pill extends Control [ height = 44, padding = [0, 16, 0, 16], fill = #DDDDDD,
    label: string = "",
    accessibilityLabel = { label },
    TextLabel [ text = { classroot.label } ]
    ]
App [ width = 300, height = 100, fill = white,
    p: Pill [ x = 20, y = 20, label = "Log a session" ]
    ]
