/*
# 4. Timer

A gauge of the elapsed time, the elapsed time in seconds, a slider for the
duration, and Reset.

`elapsed` advances on a frame clock while it is below `duration` — the
clock's `running` is that condition, so the timer stops by itself when it
reaches the duration and starts again by itself when the slider raises the
duration past it. Reset writes zero, and the same condition restarts it.
The gauge, the label and the slider all read the two numbers.
*/

class TimerTask [
    elapsed: number = 0,        // seconds
    duration: number = 15,      // seconds

    clock: Time [ tick = frame,
        running = { classroot.elapsed < classroot.duration },
        onTick(dt: number) { classroot.elapsed = Math.min(classroot.duration, classroot.elapsed + dt) }
        ],

    form: View [ x = center, y = center, width = 360,
        layout: SimpleLayout [ axis = y, spacing = 14 ],
        View [ width = 100%,
            layout: SimpleLayout [ axis = x, spacing = 12, align = center ],
            Text [ width = 100, text = "Elapsed Time:" ],
            ProgressBar [ flexes = true, accessibilityLabel = "Elapsed time",
                max = { classroot.duration }, value = { Math.min(classroot.elapsed, classroot.duration) } ]
            ],
        Text [ numeralWidth = tabular, padding = [0, 0, 0, 112], text = { classroot.elapsed.toFixed(1) + "s" } ],
        View [ width = 100%,
            layout: SimpleLayout [ axis = x, spacing = 12, align = center ],
            Text [ width = 100, text = "Duration:" ],
            Slider [ flexes = true, accessibilityLabel = "Duration", min = 1, max = 60, step = 0.5,
                value <-> classroot.duration ],
            Text [ width = 48, textAlign = end, numeralWidth = tabular, text = { classroot.duration.toFixed(1) + "s" } ]
            ],
        Button [ width = 100%, label = "Reset",
            onClick() { classroot.elapsed = 0 }
            ]
        ]
    ]
