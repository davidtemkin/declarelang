schema R2 [ id: number, f: string ]
class Store extends Dataset [ schema = [ rows[]: R2 ],
    n: number = 3,
    _m: Map<string, number> = { new Map() },
    onInit() {
        const rows: R2[] = []
        for (let r = 0; r < n; r++) rows.push(({ id: r, f: "" }))
        this.set([], ({ rows: rows }))
        },
    put(k: string) { _m.set(k, 1); this.set(["rows", 0, "f"], k) },
    keys() -> number { return _m.size }
    ]

App [ width = 400, height = 200,
    s: Store { { "rows": [] } }
    ]
