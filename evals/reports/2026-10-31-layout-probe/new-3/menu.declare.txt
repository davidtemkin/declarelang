        },

    menu: View [ x = 20, y = 20, padding = 4, cornerRadius = 6, fill = white,
        textColor = black, datapath = { entries.value },

        View [ datapath = :items[], padding = [4, 12, 4, 12], cornerRadius = 4,
            fill = { hovered ? 0xDDE6F5 : 0xFFFFFF },

    ]
