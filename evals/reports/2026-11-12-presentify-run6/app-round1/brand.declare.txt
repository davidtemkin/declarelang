/*
# Presentify: the look

Everything that makes Presentify look like itself, kept apart from how it
works: the palette (a theme laid over the library's), the drawn icon set,
the wordmark, and the small controls the screens are built from.

The identity, in a line: warm paper chrome, ink-dark type, and one violet —
"Presentify violet" — for everything the person is holding (the selection,
its handles, the current slide, the main action), with a coral spark for
the brand's own moments (the wordmark, an empty state). The slide is always
the brightest thing on screen; the chrome stays quiet around it.
*/

theme Presentify [

    background = #F4F1EC, surface = #FFFFFF, card = #FFFFFF, chrome = #FBF9F6, well = #E9E5DE,
    line = #E3DED6, lineStrong = #CFC8BD,

    hover = #F0ECE6, pressedFill = #E6E1D9,

    accentSoft = #ECE8FF, accentLine = #C9BEFF,
    spark = #FF6B4A, sparkSoft = #FFE4DC, butter = #FFC94D, mint = #2BB59A,

    // states

    danger = #D63B2F, dangerSoft = #FDE9E7, warn = #B86E00, warnSoft = #FFF4DC, ok = #2E9A6A,

class Glyph extends Icon [

            case "group": d.setLineDash([1.6, 1.8]); d.roundRect(1.8, 1.8, 12.4, 12.4, 2); d.stroke(); d.setLineDash([]); d.beginPath()
                d.roundRect(4.5, 4.5, 4, 4, 0.8); d.roundRect(7.5, 7.5, 4, 4, 0.8); d.stroke(); break

// The mark: a slide-shaped card in violet, tilted, with a coral spark — a
// presentation about to begin. Drawn, so it is crisp at every size.

class Logo [ width = 30, height = 30,
    draw(d: DrawContext) {
        const s = d.w / 30
        d.scale(s, s)
        d.save()
        d.translate(15, 15); d.rotate(-0.16); d.translate(-15, -15)
        d.beginPath(); d.roundRect(3, 6.5, 24, 17, 4.5); d.fillStyle = provided("theme").accent; d.fill()
        d.beginPath(); d.moveTo(12.5, 10.5); d.lineTo(20.5, 15); d.lineTo(12.5, 19.5); d.closePath()
        d.fillStyle = 0xFFFFFF; d.fill()
        d.restore()
        d.beginPath(); d.arc(25.5, 6, 3.4, 0, Math.PI * 2); d.fillStyle = provided("theme").spark; d.fill()
        }
    ]


class Wordmark [
    layout: SimpleLayout [ axis = x, spacing = 9, align = center ],
    Logo [ ],
    Text [ fontFamily = { app.brandFace }, fontSize = 21, fontWeight = 800, letterSpacing = -0.6,
        textColor = { provided("theme").text }, text = "presentify" ]
    ]


// ── Controls ────────────────────────────────────────────────────────────────
//
// Something a person presses is a Control: focus, Space and Enter, hover and
// press come with it. Each says what it does to a pointer resting on it
// (`tooltipLabel`) and to a screen reader (`accessibilityLabel`).

// An icon button: the toolbar's tools and commands, the inspector's arrange row.

class IconTool extends Control [ width = 34, height = 34, cornerRadius = 9,
    shape: string = "",
    label: string = "",
    on: boolean = false,
    tooltipLabel = { classroot.label },
    accessibilityLabel = { classroot.label },
    fill = { pressed ? provided("theme").pressedFill : hovered ? provided("theme").hover : null },
    State [ applied = { classroot.on }, fill = { provided("theme").accentSoft } ],
    opacity = { disabled ? 0.38 : 1 },
    Glyph [ x = center, y = center, iconSize = 17, weight = 1.7, shape = { classroot.shape },
        ink = { classroot.on ? provided("theme").accent : provided("theme").text } ]
    ]


// A tool: one of the five, pressed in while it is the one in hand.

class ToolToggle extends IconTool [ accessibilityRole = toggle, accessibilityPressed = { classroot.on } ]

// A labelled pill button: "Present", "New presentation".

class Pill extends Control [ height = 34, cornerRadius = 17, padding = [0, 16, 0, 12],
    label: string = "",
    shape: string = "",
    primary: boolean = false,
    accessibilityLabel = { classroot.label },
    fill = { classroot.primary ? (pressed ? provided("theme").accentPressed : hovered ? provided("theme").accentHovered : provided("theme").accent)
        : (pressed ? provided("theme").pressedFill : hovered ? provided("theme").hover : provided("theme").surface) },
    stroke = { classroot.primary ? null : stroke(1, provided("theme").line) },
    shadow = { classroot.primary ? shadow(0, 2, 8, 0x5B3DF540) : null },
    scale = { pressed ? 0.97 : 1 }, pivotX = { this.width / 2 }, pivotY = 17,
    opacity = { disabled ? 0.45 : 1 },
    layout: SimpleLayout [ axis = x, spacing = 7, align = center ],
    Glyph [ visible = { classroot.shape != "" }, iconSize = 15, weight = 1.8, shape = { classroot.shape },
        ink = { classroot.primary ? 0xFFFFFF : provided("theme").text } ],
    TextLabel [ fontSize = 14, fontWeight = 600, wrap = false,
        textColor = { classroot.primary ? 0xFFFFFF : provided("theme").text }, text = { classroot.label } ]
    ]


// A pop-up button: the current choice and a chevron; its menu opens under it.

// A colour well: the colour it holds, "none" as a slashed circle, or a
// mixed selection as a split chip. Pressing opens the palette for it.

class ColorWell extends Control [ width = 44, height = 30, cornerRadius = 8,
    role: string = "",             // which colour this is: fill, stroke, words, background
    label: string = "",
    value: string = "",            // "#RRGGBB", "none", or "mixed"
    tooltipLabel = { classroot.label },
    accessibilityLabel = { classroot.label + ": " + (classroot.value == "mixed" ? "mixed" : classroot.value == "none" ? "none" : classroot.value) },
    fill = { hovered ? provided("theme").hover : provided("theme").surface },
    stroke = { stroke(1, provided("theme").lineStrong) },
    opacity = { disabled ? 0.38 : 1 },
    press() { app.palette.showFor(this) },
    chip: View [ x = 4, y = 4, width = 36, height = 22, cornerRadius = 5, clip = true,
        fill = { classroot.value.startsWith("#") ? hex(classroot.value) : null },
        stroke = { classroot.value.startsWith("#") ? stroke(1, 0x00000018) : null },
        Glyph [ visible = { classroot.value == "none" }, x = center, y = center, iconSize = 15, weight = 1.5,

        TextLabel [ visible = { classroot.value == "mixed" }, x = center, fontSize = 11, fontWeight = 600,

    ]

// A number field with a unit, kept by the selection: it shows "–" when the
// selection's values differ, and a change applies to them all.

class NumField extends TextInput [ width = 64, height = 30, fontSize = 13, numeralWidth = tabular,
    padding = [0, 8, 0, 8], cornerRadius = 7, commitOn = "end",
    num: number = 0,
    mixed: boolean = false,
    value = { classroot.mixed || classroot.disabled ? "" : String(Math.round(classroot.num * 10) / 10) },
    placeholder = { classroot.mixed ? "Mixed" : "" },
    admit(t: string) -> string | null { return /^-?[0-9]*\.?[0-9]*$/.test(t) ? t : null },

    // Each use says where the number goes, in its own `commit`.

    commit(t: string) { }
    ]
