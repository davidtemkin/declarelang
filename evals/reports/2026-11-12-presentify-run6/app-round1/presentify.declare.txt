/*
# Presentify

A slide editor: presentations made of slides, shapes and words, kept by the
presentation service on port 8360 (`task/api/API.md`), and given full screen.

How it is put together:

- **The records** — the service holds every presentation. The one open is
  copied into `app.doc` (a `Doc`, model.declare), the working copy every
  view reads and every edit writes. `app.saver` (sync.declare) carries the
  difference back, unasked, within a moment of each pause, and says so.
- **Where the person is** — the App's `location` is the open presentation's
  id ("" for the list), so Back returns to the list. Inside a presentation,
  `app.ed` (an `Editor`, editing.declare) holds the slide, the selection, the
  tool and the text being typed, and every verb that changes them; `app.history`
  keeps each verb as one undoable step.
- **The screens** — `HomeScreen` (home.declare), the list; `EditorScreen`,
  below: `TopBar`, `SlideRail` and `Inspector` (panels.declare) around the
  `Stage` (stage.declare); and `Presenter`, below. Every slide anywhere is a
  `SlideView` (slide.declare), so the editor, the pictures and presenting
  always agree.
- **The look** — brand.declare: the palette, the icons, the small controls.

Opening a presentation grows it from its card (one spring, `openT`, that the
editor's place and scale read); closing shrinks it back.
*/

    // Presenting fills the screen. The browser ends full screen itself on
    // Escape, so it says when it has (`onExit`).
    function enterFullScreen(onExit: () => void) {
        const el = document.documentElement
        if (el.requestFullscreen == null) return
        el.requestFullscreen().then(() => {
            const watch = () => { if (document.fullscreenElement == null) { document.removeEventListener("fullscreenchange", watch); onExit() } }
            document.addEventListener("fullscreenchange", watch)
        }).catch(() => { })
    }
    // Closing the window while something is not yet saved asks first.
    function guardLeaving(unsaved: () => boolean) {
        window.addEventListener("beforeunload", (e) => { if (unsaved()) { e.preventDefault(); e.returnValue = "" } })
    }
    function leaveFullScreen() { if (document.fullscreenElement != null) document.exitFullscreen().catch(() => { }) }
    }

        history.clear()

        saver.start(deck, slides)

                this.url = app.saver.api + "/decks/" + this._deck.id + "/slides"
                this.fetch()
                return

            this._deck = null

            },

            this._deck = null

        if (saver.status != "saved" && saver.deckId != "") { saver.flush() }
        loader._deck = null
        loader.url = saver.api + "/decks/" + deckId
        loader.fetch()
        },

    // ── presenting

    present() {
        if (!open || loadingDeck) return
        ed.finishWork()
        showAt = Math.max(0, ed.slideIndex)
        hintGone = false
        presenting = true
        enterFullScreen(() => app.endPresenting())
        afterDelay(2600, () => { app.hintGone = true })
        },
    endPresenting() {
        if (!presenting) return
        const at = app.presenter?.at ?? showAt
        presenting = false
        leaveFullScreen()
        const s = app.doc.slides()[at]
        if (s != null) ed.go(s.id)
        afterSettle(() => { if (app.stage != null) Focus.focus(app.stage) })
        },

            "Times New Roman", "American Typewriter", "Menlo"].map((f) => ({ id: f, label: f, checked: this.current == f })) },
        picked(id: string) { app.ed.setWords("Font", ({ font: id })) }
        ],
    weightMenu: Menu [
        items = { [[300, "Light"], [400, "Regular"], [500, "Medium"], [600, "Semibold"], [700, "Bold"], [800, "Heavy"]]
            .map((w) => ({ id: "" + w[0], label: "" + w[1] })) },
        picked(id: string) { app.ed.setWords("Weight", ({ weight: Number(id) })) }
        ],

    palette: Palette [ ],

    // ── asked first: renaming and deleting a presentation

    renaming: string = "",
    askRename(id: string, title: string) {
        renaming = id
        renameDialog.draft = title
        renameDialog.show()
        afterDelay(60, () => app.renameDialog.field.select("all"))
        },
    renameDialog: Dialog [ title = "Rename presentation",
        draft: string = "",
        buttons = { [ ({ id: "cancel", label: "Cancel", cancel: true }), ({ id: "ok", label: "Rename", default: true }) ] },
        picked(id: string) { if (id == "ok") app.home.rename(app.renaming, this.draft) },
        field: TextInput [ width = 100%, accessibilityLabel = "Title", value <-> app.renameDialog.draft ]
        ],
    askDelete(id: string, title: string) {
        confirm.open("Delete “" + (title == "" ? "Untitled presentation" : title) + "”?",
            "The presentation and all of its slides will be deleted. This can't be undone.",
            [ ({ id: "cancel", label: "Cancel", cancel: true }), ({ id: "delete", label: "Delete", default: true }) ],
            "error",
            (choice: string) => { if (choice == "delete") app.home.remove(id) })
        },

    // Deleting is said in the danger colour.

    confirm: Dialog [ theme = { ({ ...provided("theme"), accent: provided("theme").danger,
        accentHovered: 0xC0352A, accentPressed: 0xA82E24 }) } ],

    toast: Toast [ duration = 5000 ]
    ]
