/*
# Presentify: the panels around the slide

- **`SlideRail`** — the slides in order down the left, each a live picture
  of its slide. A press opens one; the arrow keys move between them; a drag
  reorders them, and the others glide into their new places when it lands.
- **`TopBar`** — the way home, the title, whether everything is saved, the
  tools, undo and redo, and Present.
- **`Inspector`** — the selection's fill, outline, corners and text, and
  arranging. Its shape never changes: a section that doesn't apply to what
  is selected stays where it is, quieted, and says why.
*/

// Stacks the rows, and glides them when they reorder, arrive or leave.

class RailLayout extends TweenLayout [ duration = 260,
    rowH: number = 104,
    place() {
        const w = this.contentExtent("width")
        return this.laid().map((c, i) => ({ x: 0, y: i * this.rowH, w: w, h: this.rowH, vis: true }))
        }
    ]

// The drag that reorders the list: where it started, where it would land.

class RailDrag extends Node [
    dragId: string = "",
    dropAt: number = 0,
    ghostY: number = 0,
    _from: number = 0,
    _y0: number = 0,
    start(id: string, rowTop: number) {
        dragId = id
        _from = app.doc.indexOf(id)
        _y0 = rowTop
        ghostY = rowTop + 8
        dropAt = _from
        },
    follow(dy: number) {
        ghostY = _y0 + 8 + dy
        const n = app.doc.slides().length
        dropAt = Math.max(0, Math.min(n, Math.round((_y0 + dy) / 104 + 0.0)))
        },
    finish(canceled: boolean) {
        const id = dragId
        dragId = ""
        if (canceled || id == "") return
        const to = dropAt > _from ? dropAt - 1 : dropAt
        app.ed.moveSlide(id, to)
        }
    ]


class SlideRow extends Control [ width = 100%, height = 104,
    slideId: string = { :id ?? "" },
    here: boolean = { app.ed.currentId == classroot.slideId },
    carried: boolean = { app.rail.dragId == classroot.slideId },
    accessibilityLabel = { "Slide " + (rowIndex + 1) },
    accessibilityRole = option,
    accessibilitySelected = { classroot.here },
    press() { app.ed.go(classroot.slideId) },
    onDragStart(e: PointerEvent) { app.ed.go(classroot.slideId); app.rail.start(classroot.slideId, this.y) },
    onDrag(e: PointerEvent) { app.rail.follow(e.deltaY) },
    onDragEnd(e: DragEndEvent) { app.rail.finish(e.canceled) },
    onContextMenu(e: PointerEvent) { app.ed.go(classroot.slideId); app.slideMenu.open(this, e) },
    focusShape() -> object { return this.frame },
    opacity = { classroot.carried ? 0.35 : 1 },

    num: Text [ x = 8, y = 10, width = 18, textAlign = right, fontSize = 12, fontWeight = 600, numeralWidth = tabular,
        textColor = { classroot.here ? provided("theme").accent : provided("theme").textFaint },
        text = { "" + (classroot.rowIndex + 1) } ],
    frame: View [ x = 30, y = 8, width = 160, height = 90, cornerRadius = 6, clip = true,
        stroke = { classroot.here ? stroke(2.5, provided("theme").accent)
            : classroot.hovered ? stroke(1, provided("theme").lineStrong) : stroke(1, provided("theme").line) },
        shadow = { classroot.here ? shadow(0, 4, 12, 0x5B3DF530) : shadow(0, 1, 2, 0x1E1B2E14) },
        SlideView [ scale = 0.125, pivotX = 0, pivotY = 0 ]
        ]
    ]

// The colour palette, opened by any well: what it sets follows the well's `role`.

class Palette extends Popover [ placement = "below",
    role: string = { (this.opener as ColorWell)?.role ?? "" },
    current: string = { (this.opener as ColorWell)?.value ?? "" },
    pick(c: string?) { app.setColour(this.role, c) },
    layout: SimpleLayout [ axis = y, spacing = 12 ],
    Text [ fontSize = 12, fontWeight = 700, textColor = { provided("theme").textMuted },
        text = { classroot.role == "background" ? "Background" : classroot.role == "stroke" ? "Outline"
            : classroot.role == "words" ? "Text colour" : "Fill" } ],
    grid: View [ width = 232, datapath = { app.swatches.value },
        layout: WrappingLayout [ spacing = 8, rowSpacing = 8 ],
        Swatch [ datapath = :colors[], color = { :@ ?? "" }, picked = { classroot.current == (:@ ?? "") },
            press() { classroot.pick(:@ ?? "") ; classroot.close() }
            ]
        ],
    View [ width = 232,
        layout: SimpleLayout [ axis = x, spacing = 8, align = center ],
        Swatch [ visible = { classroot.role == "fill" || classroot.role == "stroke" }, color = "",
            picked = { classroot.current == "none" },
            press() { classroot.pick(null); classroot.close() }
            ],
        Text [ visible = { classroot.role == "fill" || classroot.role == "stroke" }, fontSize = 12,
            textColor = { provided("theme").textMuted }, text = "None" ],
        Spacer [ ],
        TextInput [ width = 104, height = 30, fontSize = 13, accessibilityLabel = "Hex colour", placeholder = "#RRGGBB",
            commitOn = "end",
            value = { classroot.current.startsWith("#") ? classroot.current : "" },
            admit(t: string) -> string | null { return /^#?[0-9a-fA-F]{0,6}$/.test(t) ? t : null },
            commit(t: string) {
                const v = (t.startsWith("#") ? t : "#" + t).toUpperCase()
                if (/^#[0-9A-F]{6}$/.test(v)) classroot.pick(v)
                }
            ]
        ]
    ]
