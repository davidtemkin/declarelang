/*
# Ledger: the screens

The shell (navigation, the filter bar) and the four pages. Each page holds
the reads it needs as members — created when the page opens, addressed by
the App's filter (`app.fq`) and the write revision (`app.rev`) — so a
page's figures always describe the same records as the list.
*/

// A read of the service whose address follows the filters. Nothing is sent
// until the lookups (and so today's date) have arrived.

class Read extends DataSource [ autoFetch = true,
    path: string = "",
    url = { app.ready && classroot.path != "" ? app.base + classroot.path + (classroot.path.includes("?") ? "&" : "?") + "rev=" + app.rev : "" },

    // an answer that takes more than a moment says the service is still working

    slow: boolean = false,
    trackChanges = ["loading"],
    onChange() {
        if (!this.loading) { slow = false; return }
        afterDelay(1500, () => { classroot.slow = classroot.loading })
        }
    ]

class FilterBar [
    layout: SimpleLayout [ axis = y, spacing = 8 ],

    // the active filters as chips: what each says, and the filter without it

    chipData: Dataset [ contents = { { chips: classroot.chipList() } } ],
    chips: array = { this.chipData.value?.chips ?? [] },
    chipList() -> array {
        const out = []
        const all = (x: FilterSpec) => filterQuery(x)
        const base = app.spec(app.from, app.to)
        if (app.fQ.trim() != "") out.push(({ id: "q", label: "Search “" + app.fQ.trim() + "”", without: all(({ ...base, q: "" })) }))
        if (app.fMin >= 0 || app.fMax >= 0) out.push(({ id: "amount", label: classroot.amountLabel(), without: all(({ ...base, min: -1, max: -1 })) }))
        if (app.fKinds.length > 0) out.push(({ id: "kind", label: app.fKinds.map((k) => k == "sale" ? "Sales" : k == "refund" ? "Refunds" : "Expenses").join(" or "), without: all(({ ...base, kinds: [] })) }))
        if (app.fCats.length > 0) out.push(({ id: "cat", label: app.fCats.map((c) => c == "none" ? "Uncategorised" : app.books.catName(c)).join(" or "), without: all(({ ...base, cats: [] })) }))
        if (app.fContacts.length > 0) out.push(({ id: "contact", label: app.fContacts.map((c) => c == "none" ? "No contact" : app.books.contactName(c)).join(" or "), without: all(({ ...base, contacts: [] })) }))
        if (app.fProducts.length > 0) out.push(({ id: "product", label: app.fProducts.map((p) => p == "none" ? "Other items" : app.books.productName(p)).join(" or "), without: all(({ ...base, products: [] })) }))
        return out
        },
    amountLabel() -> string {
        if (app.fMax < 0) return "≥ " + moneyWhole(app.fMin)
        if (app.fMin <= 0) return "≤ " + moneyWhole(app.fMax)
        return moneyWhole(app.fMin) + " – " + moneyWhole(app.fMax)
        },
    remove(id: string) {
        if (id == "q") app.fQ = ""
        else if (id == "amount") { app.fMin = -1; app.fMax = -1 }
        else if (id == "kind") app.fKinds = []
        else if (id == "cat") app.fCats = []
        else if (id == "contact") app.fContacts = []
        else if (id == "product") app.fProducts = []
        },

    // the controls: dates, comparison, search, and the filter menus

    controls: View [ width = 100%,
        layout: WrappingLayout [ spacing = 8, rowSpacing = 8, align = center ],
        dates: View [ height = { app.touchFirst ? 44 : 32 }, cornerRadius = 8, fill = { provided("theme").control },
            layout: SimpleLayout [ axis = x, align = center ],
            Control [ width = { app.touchFirst ? 44 : 30 }, height = { parent.height }, cornerRadius = 8, accessibilityLabel = "Previous range", tooltipLabel = "Previous range  [",
                fill = { hovered ? provided("theme").controlHovered : null },
                press() { app.stepDates(-1) },
                TextLabel [ x = center, fontSize = 14, text = "‹" ]
                ],
            DropButton [ fill = null, label = { app.rangeName }, accessibilityLabel = { "Date range: " + app.rangeName + ", " + rangeLabel(app.from, app.to) },
                tooltipLabel = { rangeLabel(app.from, app.to) },
                press() { app.dateMenu.showFor(this) }
                ],
            Control [ width = { app.touchFirst ? 44 : 30 }, height = { parent.height }, cornerRadius = 8, accessibilityLabel = "Next range", tooltipLabel = "Next range  ]",
                fill = { hovered ? provided("theme").controlHovered : null },
                press() { app.stepDates(1) },
                TextLabel [ x = center, fontSize = 14, text = "›" ]
                ]
            ],
        DropButton [ label = { app.fCompare == "none" ? "No comparison" : "vs " + (app.fCompare == "year" ? "last year" : "previous") },
            accessibilityLabel = { "Compare with: " + (app.fCompare == "none" ? "nothing" : app.cmpName) },
            menu = { app.compareMenu } ],
        search: SearchField [ width = { app.phone ? Math.max(140, parent.width - 16) : 220 }, height = { app.touchFirst ? 44 : 32 },
            placeholder = "Search  /", accessibilityLabel = "Search transactions",
            value = { app.fQ },
            commit(v: string) { app.fQ = v }
            ],
        amountBtn: DropButton [ label = { app.fMin >= 0 || app.fMax >= 0 ? classroot.amountLabel() : "Amount" }, on = { app.fMin >= 0 || app.fMax >= 0 },
            press() { app.amountPop.showFor(this) }
            ],
        DropButton [ label = { app.fKinds.length == 0 ? "Kind" : app.fKinds.length + " kinds" }, on = { app.fKinds.length > 0 },
            menu = { app.kindMenu } ],
        catBtn: DropButton [ label = { app.fCats.length == 0 ? "Category" : app.fCats.length == 1 ? (app.fCats[0] == "none" ? "Uncategorised" : app.books.catName(app.fCats[0])) : app.fCats.length + " categories" },
            on = { app.fCats.length > 0 },
            press() { app.catPop.showFor(this) }
            ],
        contactBtn: DropButton [ label = { app.fContacts.length == 0 ? "Contact" : app.fContacts.length == 1 ? app.books.contactName(app.fContacts[0]) || "No contact" : app.fContacts.length + " contacts" },
            on = { app.fContacts.length > 0 },
            press() { app.contactPop.showFor(this) }
            ]
        ],

    // the chips, and a word that the figures are filtered

    chipRow: View [ width = 100%, visible = { classroot.chips.length > 0 },
        layout: WrappingLayout [ spacing = 6, rowSpacing = 6, align = center ],
        Text [ fontSize = 12.5, fontWeight = semibold, textColor = { provided("theme").warn }, text = "Filtered:" ],
        View [ datapath = { classroot.chipData.value },
            layout: WrappingLayout [ spacing = 6, rowSpacing = 6 ],
            FilterChip [ datapath = :chips[], label = { "" + :label },
                remove() { classroot.remove("" + :id) }
                ]
            ],
        QuietButton [ label = "Clear all", tone = { provided("theme").accent },
            press() { app.clearAll() }
            ]
        ]
    ]

// ── Cells: each a view over its row's record ────────────────────────────────
//
// An editable cell is a control showing its value; Enter (or a press)
// opens a field in its place. In the field, Return saves, Escape cancels
// and Tab saves and moves on.

class EditCell [ width = 100%, height = 100%,
    editing: boolean = false,
    canEdit: boolean = true,
    txn: Txn = { :@ },

    // the text the field starts from, and what to do with the text it ends with

    start: string = "",
    label: string = "",
    accept(text: string) { },
    begin() {
        if (!canEdit) return
        app.editStartedAt = Date.now()
        editing = true
        afterSettle(() => { if (classroot.field != null) Focus.focus(classroot.field) })
        },
    stop() { editing = false; Focus.focus(this.show) },
    defaultPlacement = show,
    show: Control [ visible = { !classroot.editing }, width = 100%, height = 100%, padding = [0, 8, 0, 8],
        accessibilityLabel = { classroot.label + (classroot.canEdit ? ". Press Enter to edit." : "") },

        // a click picks the row, as anywhere in it; a double-click, Return or F2 edits

        onClick(e: PointerEvent) { app.rowPressed(classroot.txn, e) },
        onDoubleClick() { classroot.begin() },
        press() { classroot.begin() }
        ],
    field: TextInput [ exists = { classroot.editing }, x = 2, y = center, width = { parent.width - 4 }, height = 30,
        fontSize = 13, commitOn = "end",
        accessibilityLabel = { classroot.label },
        value = { classroot.start },

        // the whole value is selected, so typing replaces it

        onFocus() { super.onFocus(); afterDelay(0, () => this.select("all")) },
        commit(v: string) { if (v != classroot.start) classroot.accept(v) },
        onEnter() { classroot.stop() },

        // leaving saves: the edit ends here, before the field is put away

        onBlur() { super.onBlur(); this.endEdit(); classroot.editing = false },
        onKeyDown(e: KeyEvent) {
            if (e.key == "Escape") { this.cancelEdit(); classroot.stop(); return }
            super.onKeyDown(e)
            }
        ]
    ]


class DateCell extends EditCell [
    label = { "Date, " + fmtDate(classroot.txn.date) },
    start = { fmtDate(classroot.txn.date, app.books.year) },
    pending: string = { app.unsaved(classroot.txn, "date") },
    accept(text: string) {
        const iso = parseDate(text, app.today)
        if (iso == "") { app.toast.say("“" + text + "” isn't a date — try “3 nov” or “yesterday”"); return }
        if (iso != txn.date) app.saveEdit(txn, "date", iso, fmtDate(iso))
        },
    Text [ visible = { !classroot.editing }, y = center, fontSize = 13, wrap = false, numeralWidth = tabular,
        textColor = { classroot.pending != "" ? provided("theme").bad : provided("theme").text },
        text = { classroot.pending != "" ? classroot.pending : fmtDate(classroot.txn.date, app.books.year) } ]
    ]

// The phone's list: grouped by day, each row its description, contact,
// amount and category. Tapping a row opens it.

schema PhoneLine [ id: string, head?: string, total?: number, txn?: Txn ]

class PhoneList [ scrolls = y,
    opacity = { app.txns.loading && app.txns.loaded ? 0.55 : 1 },
    lines: Dataset [ schema = [ lines[]: PhoneLine ], contents = { classroot.group(app.rows) } ],
    group(rows: Txn[]) -> object {
        const out = []
        let day = ""
        for (const t of rows) {
            if (app.sortOn == "date" && t.date != day) { day = t.date; out.push(({ id: "d" + day, head: fmtDay(day, app.today) })) }
            out.push(({ id: t.id, txn: t }))
            }
        return ({ lines: out })
        },
    datapath = { this.lines.value },
    layout: SimpleLayout [ axis = y ],
    View [ datapath = :lines[], virtualize = true, classFor = { :head != null ? DayHead : PhoneRow } ]
    ]
