/*
# Ledger: Overview, Insights and Budgets

The pages that answer "how are we doing". Each figure is one the service
computed over the App's filter, and each opens the records behind it
through `app.drill(…)`, which sets the filters to exactly those records —
their dates, kinds and group — so the list's summary equals the figure.
*/

// The measures a trend can show, and what opening one of its points opens

schema Measure2 [ id: string, name: string, unit: "money" | "percent" | "count", kinds[]: string, up: string ]

// A filtered banner: every screen says when its figures are filtered

    QuietButton [ label = "Clear filters", tone = { provided("theme").accent },

        ]

// A chart's frame: title, one plain sentence, the chart or its table

class ChartFrame extends Panel [
    sentence: string = "",
    source: Read? = null,
    tableShown: boolean = false,

    // the toggle sits in the panel's own heading

    Text [ width = 100%, fontSize = 13, textColor = { provided("theme").textMuted }, wrap = true, accessibilityLive = polite,
        text = { classroot.sentence } ],
    LoadState [ source = { classroot.source } ]
    ]


// One headline figure: its value, its change with an arrow and words, a
// small line of the last periods. Pressing opens its trend in Insights.

class Kpi extends Control [ flexes = true, height = 128, cornerRadius = 12, padding = 14,
    label: string = "",
    figure: string = "",
    change: number? = null,          // percent, or points for a percentage
    points: boolean = false,
    upIsGood: boolean = true,
    vs: string = "",
    spark: number[] = [],
    tone: Color = gray,
    good: boolean = { classroot.change != null && classroot.change != 0 && (classroot.change > 0) == classroot.upIsGood },
    changeText: string = { classroot.describe(classroot.change) },

    // "▲ 12% vs Oct", or for a percentage "▼ −4.2 pts vs Oct"

    describe(c: number?) -> string {
        if (c == null) return vs == "" ? "" : "No figure for " + vs
        const arrow = c > 0 ? "▲ " : c < 0 ? "▼ " : ""
        const size = points ? (c > 0 ? "+" : c < 0 ? "−" : "") + Math.abs(Math.round(c * 10) / 10) + " pts" : signedPct(Math.abs(c)).replace("+", "")
        return arrow + size + " vs " + vs
        },
    accessibilityLabel = { classroot.label + ", " + classroot.figure + (classroot.changeText != "" ? ", " + classroot.changeText.replace("▲", "up").replace("▼", "down") : "") + ". Opens its trend." },
    fill = { hovered || pressed ? provided("theme").rowHover : provided("theme").surface },
    stroke = { stroke(1, provided("theme").line) },
    layout: SimpleLayout [ axis = y, spacing = 4 ],
    Text [ fontSize = 13, textColor = { provided("theme").textMuted }, text = { classroot.label } ],
    Text [ fontSize = { app.phone ? 20 : 24 }, fontWeight = semibold, wrap = false, numeralWidth = tabular, text = { classroot.figure } ],
    Text [ fontSize = 12, wrap = false, visible = { classroot.changeText != "" },
        textColor = { classroot.change == null || classroot.change == 0 ? provided("theme").textMuted : classroot.good ? provided("theme").good : provided("theme").bad },
        text = { classroot.changeText } ],
    Sparkline [ width = 100%, height = 26, values = { classroot.spark }, tone = { classroot.tone } ]
    ]


// ── Overview ────────────────────────────────────────────────────────────────

class OverviewPage [ scrolls = y,
    // the figures for the range and for the comparison period

    now: Read [ schema = Periods, path = { "/api/aggregates/periods?" + app.fq + "&by=month" } ],
    prev: Read [ schema = Periods, path = { app.cmp == null ? "" : "/api/aggregates/periods?" + app.fqCompare + "&by=month" } ],

    // the last twelve months to the end of the range, for the small lines and the chart

    end: string = { monthEnd(app.to > app.today ? app.today : app.to) },
    trend: Read [ schema = Periods, path = { "/api/aggregates/periods?" + filterQuery(app.spec(addMonths(monthStart(classroot.end), -11), classroot.end)) + "&by=month" } ],
    cats: Read [ schema = CategoryTotals, path = { "/api/aggregates/categories?" + app.fq + "&side=expense" } ],
    prods: Read [ schema = ProductTotals, path = { "/api/aggregates/products?" + app.fq } ],
    custs: Read [ schema = ContactTotals, path = { "/api/aggregates/contacts?" + app.fq + "&side=income&limit=5" } ],
    uncat: Read [ schema = Summary, path = "/api/summary?category=none" ],
    month: string = { (app.to > app.today ? app.today : app.to).slice(0, 7) },
    budgets: Read [ schema = Budgets, path = { "/api/budgets?month=" + classroot.month } ],
    recent: Read [ schema = TxnPage, path = { "/api/transactions?" + app.fq + "&limit=8" } ],

    tot: PeriodTotal? = { this.now.value?.total ?? null },
    was: PeriodTotal? = { this.prev.value?.total ?? null },
    months: Period[] = { this.trend.value?.periods ?? [] },
    pct(a: number, b: number?) -> number? { return b == null ? null : changePct(a, b) },
    marginPts: number? = { classroot.was?.grossMargin.percent == null || classroot.tot?.grossMargin.percent == null ? null
        : (classroot.tot?.grossMargin.percent ?? 0) - (classroot.was?.grossMargin.percent ?? 0) },

    // the expense categories: the top five as bars, the rest as "Other"

    catRows: Dataset [ contents = { classroot.topCats() } ],
    topCats() -> object {
        const all = this.cats.value?.categories ?? []
        const top = all.slice(0, 5).map((c) => ({ id: c.categoryId ?? "none", label: app.books.catName(c.categoryId ?? null), amount: c.amount,
            tone: c.categoryId == null ? provided("theme").warn : provided("theme").cats[Math.max(0, app.books.catColour(c.categoryId))], ids: [c.categoryId ?? "none"] }))
        const rest = all.slice(5)
        if (rest.length > 0) top.push(({ id: "rest", label: "Other (" + rest.length + " categories)", amount: rest.reduce((n, c) => n + c.amount, 0),
            tone: provided("theme").compareColor, ids: rest.map((c) => c.categoryId ?? "none") }))
        const peak = Math.max(1, ...top.map((r) => r.amount))
        return ({ rows: top.map((r) => ({ ...r, peak: peak })) })
        },
    prodRows: Dataset [ contents = { classroot.topProducts() } ],
    topProducts() -> object {
        const list = (this.prods.value?.products ?? []).slice(0, 5)
        const peak = Math.max(1, ...list.map((p) => p.revenue))
        return ({ rows: list.map((p) => ({ id: p.productId ?? "none", label: app.books.productName(p.productId ?? null), amount: p.revenue, units: p.units, peak: peak })) })
        },
    custRows: Dataset [ contents = { classroot.topCustomers() } ],
    topCustomers() -> object {
        const list = this.custs.value?.contacts ?? []
        const peak = Math.max(1, ...list.map((c) => c.amount))
        return ({ rows: list.map((c) => ({ id: c.contactId ?? "none", label: c.contactName ?? "No customer named (market sales)", amount: c.amount, peak: peak })) })
        },

    // what needs attention: uncategorised records, budgets over or near

    attention: Dataset [ contents = { classroot.attentionItems() } ],
    attentionItems() -> object {
        const out = []
        const n = this.uncat.value?.count ?? 0
        if (n > 0) out.push(({ id: "uncat", label: count(n) + " uncategorised " + (n == 1 ? "transaction" : "transactions"), detail: "Give each a category so the totals are right", tone: "warn" }))
        for (const b of this.budgets.value?.budgets ?? []) {
            if (b.spent >= b.amount) out.push(({ id: "b:" + b.categoryId, label: app.books.catName(b.categoryId) + " is over budget", detail: "Over by " + money(b.spent - b.amount) + " of " + money(b.amount) + " in " + monthName(this.month + "-01"), tone: "bad" }))
            else if (b.spent >= b.amount * 0.85) out.push(({ id: "b:" + b.categoryId, label: app.books.catName(b.categoryId) + " is near its budget", detail: money(b.amount - b.spent) + " left of " + money(b.amount), tone: "warn" }))
            }
        return ({ rows: out })
        },
    openAttention(id: string) {
        if (id == "uncat") { app.drillAll(["none"]); return }
        const cat = id.slice(2)
        app.drill(this.month + "-01", monthEnd(this.month + "-01"), ["expense"], [cat], null, null)
        },

    // the chart's periods and series

    chartPeriods: ChartPeriod[] = { classroot.months.map((p) => ({ label: periodLabel(p.period, "month", false), long: periodLabel(p.period, "month", true), from: p.from, to: p.to, partial: p.partial })) },
    chartSeries: ChartSeries[] = { [
        ({ id: "revenue", name: "Revenue", kind: "bar", tone: provided("theme").inColor, values: classroot.months.map((p) => p.revenue.amount), counts: classroot.months.map((p) => p.revenue.count) }),
        ({ id: "expenses", name: "Expenses", kind: "bar", tone: provided("theme").outColor, values: classroot.months.map((p) => p.expenses.amount), counts: classroot.months.map((p) => p.expenses.count) }),
        ({ id: "net", name: "Net profit", kind: "line", tone: provided("theme").netColor, values: classroot.months.map((p) => p.net.amount), counts: classroot.months.map((p) => p.net.count) })
        ] },

    col: View [ width = { Math.min(1180, parent.width) }, padding = { app.phone ? [8, 12, 24, 12] : [4, 20, 32, 20] },
        layout: SimpleLayout [ axis = y, spacing = 16 ],
        FilteredNote [ ],
        LoadState [ source = { classroot.now } ],

        kpis: View [ width = 100%, opacity = { classroot.now.loading ? 0.6 : 1 },
            layout: WrappingLayout [ columns = { app.phone ? 2 : 4 }, spacing = 12, rowSpacing = 12 ],
            Kpi [ label = "Revenue", figure = { money(classroot.tot?.revenue.amount ?? 0) }, vs = { app.cmpName },
                change = { classroot.pct(classroot.tot?.revenue.amount ?? 0, classroot.was?.revenue.amount ?? null) },
                spark = { classroot.months.map((p) => p.revenue.amount) }, tone = { provided("theme").inColor },
                press() { app.openTrend("revenue") }
                ],
            Kpi [ label = "Expenses", figure = { money(classroot.tot?.expenses.amount ?? 0) }, vs = { app.cmpName }, upIsGood = false,
                change = { classroot.pct(classroot.tot?.expenses.amount ?? 0, classroot.was?.expenses.amount ?? null) },
                spark = { classroot.months.map((p) => p.expenses.amount) }, tone = { provided("theme").outColor },
                press() { app.openTrend("expenses") }
                ],
            Kpi [ label = "Net profit", figure = { money(classroot.tot?.net.amount ?? 0) }, vs = { app.cmpName },
                change = { classroot.pct(classroot.tot?.net.amount ?? 0, classroot.was?.net.amount ?? null) },
                spark = { classroot.months.map((p) => p.net.amount) }, tone = { provided("theme").netColor },
                press() { app.openTrend("net") }
                ],
            Kpi [ label = "Gross margin", figure = { percent(classroot.tot?.grossMargin.percent ?? null) }, vs = { app.cmpName }, points = true,
                change = { classroot.marginPts },
                spark = { classroot.months.map((p) => p.grossMargin.percent ?? 0) }, tone = { provided("theme").runColor },
                press() { app.openTrend("margin") }
                ]
            ],

        chart: ChartFrame [ title = "Income vs expenses", source = { classroot.trend },
            sentence = { app.incomeSentence(classroot.months) },
            Legend [ items = { classroot.chartSeries } ],
            plot: TimeChart [ width = 100%, height = { app.phone ? 200 : 260 }, title = "Income vs expenses, month by month",
                periods = { classroot.chartPeriods }, series = { classroot.chartSeries },
                open(p: number, s: number) { app.openPeriod(classroot.chartPeriods[p], classroot.chartSeries[s].id) },
                pickRange(a: number, b: number) { app.offerRange(classroot.chartPeriods[a].from, classroot.chartPeriods[b].to) }
                ]
            ],

        trio: View [ width = 100%,
            layout: WrappingLayout [ columns = { app.wide ? 3 : 1 }, spacing = 16, rowSpacing = 16 ],
            Panel [ flexes = true, title = "Expenses by category",
                Text [ visible = { (classroot.cats.value?.categories ?? []).length == 0 }, fontSize = 13, textColor = { provided("theme").textMuted }, text = "No expenses in this range." ],
                View [ width = 100%, datapath = { classroot.catRows.value },
                    layout: SimpleLayout [ axis = y, spacing = 2 ],
                    BarItem [ datapath = :rows[], label = { "" + :label }, value = { :amount ?? 0 }, peak = { :peak ?? 1 }, tone = { :tone ?? 0 },
                        accessibilityLabel = { :label + ", " + money(:amount ?? 0) },
                        press() { app.drill(app.from, app.to, ["expense"], (:ids ?? []) as string[], null, null) }
                        ]
                    ]
                ],
            Panel [ flexes = true, title = "Top products",
                Text [ visible = { (classroot.prods.value?.products ?? []).length == 0 }, fontSize = 13, textColor = { provided("theme").textMuted }, text = "No sales in this range." ],
                View [ width = 100%, datapath = { classroot.prodRows.value },
                    layout: SimpleLayout [ axis = y, spacing = 2 ],
                    BarItem [ datapath = :rows[], label = { :label + " · " + (:units ?? 0) + " sold" }, value = { :amount ?? 0 }, peak = { :peak ?? 1 },
                        tone = { provided("theme").inColor }, accessibilityLabel = { :label + ", " + money(:amount ?? 0) },
                        press() { app.drill(app.from, app.to, [], null, null, ["" + :id]) }
                        ]
                    ]
                ],
            Panel [ flexes = true, title = "Top customers",
                Text [ visible = { (classroot.custs.value?.contacts ?? []).length == 0 }, fontSize = 13, textColor = { provided("theme").textMuted }, text = "No sales in this range." ],
                View [ width = 100%, datapath = { classroot.custRows.value },
                    layout: SimpleLayout [ axis = y, spacing = 2 ],
                    BarItem [ datapath = :rows[], label = { "" + :label }, value = { :amount ?? 0 }, peak = { :peak ?? 1 },
                        tone = { provided("theme").accent }, accessibilityLabel = { :label + ", " + money(:amount ?? 0) },
                        press() { app.drill(app.from, app.to, ["sale", "refund"], null, ["" + :id], null) }
                        ]
                    ]
                ]
            ],

        duo: View [ width = 100%,
            layout: WrappingLayout [ columns = { app.wide ? 2 : 1 }, spacing = 16, rowSpacing = 16 ],
            Panel [ flexes = true, title = "Needs attention",
                Text [ visible = { classroot.uncat.loaded && (classroot.attention.value?.rows ?? []).length == 0 }, fontSize = 14, text = "All caught up." ],
                View [ width = 100%, datapath = { classroot.attention.value },
                    layout: SimpleLayout [ axis = y, spacing = 4 ],
                    AttentionItem [ datapath = :rows[],
                        press() { classroot.openAttention("" + :id) }
                        ]
                    ]
                ],
            Panel [ flexes = true, title = "Recent transactions",
                Text [ visible = { classroot.recent.loaded && (classroot.recent.value?.rows ?? []).length == 0 }, fontSize = 13,
                    textColor = { provided("theme").textMuted }, text = "Nothing recorded in this range yet. Record a sale or an expense to start." ],
                View [ width = 100%, datapath = { classroot.recent.value },
                    layout: SimpleLayout [ axis = y ],
                    RecentRow [ datapath = :rows[] ]
                    ]
                ]
            ]
        ]
    ]

class TrendsView [ width = 100%,
    page: InsightsPage? = null,
    measures: Measure2[] = { app.measures },
    m: Measure2 = { app.measures.find((x) => x.id == app.trendMeasure) ?? app.measures[0] },
    pick(p: Period) -> number {
        const id = this.m.id
        return id == "revenue" ? p.revenue.amount : id == "expenses" ? p.expenses.amount : id == "net" ? p.net.amount
            : id == "margin" ? (p.grossMargin.percent ?? 0) : p.sales.count
        },
    pickCount(p: Period) -> number {
        const id = this.m.id
        return id == "revenue" ? p.revenue.count : id == "expenses" ? p.expenses.count : id == "net" ? p.net.count
            : id == "margin" ? p.grossMargin.count : p.sales.count
        },
    series: ChartSeries[] = { [
        ({ id: "now", name: classroot.m.name, kind: "line" as const, tone: provided("theme").accent,
           values: (classroot.page?.periods ?? []).map((p) => classroot.pick(p)), counts: (classroot.page?.periods ?? []).map((p) => classroot.pickCount(p)) }),
        ...(app.cmp == null ? [] : [({ id: "was", name: classroot.m.name + ", " + (app.fCompare == "year" ? "a year earlier" : "the period before"), kind: "dashed" as const, tone: provided("theme").compareColor,
           values: (classroot.page?.periods ?? []).map((p, i) => { const q = (classroot.page?.prevPeriods ?? [])[i]; return q == null ? 0 : classroot.pick(q) }),
           counts: (classroot.page?.periods ?? []).map((p, i) => { const q = (classroot.page?.prevPeriods ?? [])[i]; return q == null ? 0 : classroot.pickCount(q) }) })])
        ] },
    sentence: string = { app.peakSentence(classroot.m.name, classroot.series[0]?.values ?? [], classroot.page?.chartPeriods ?? [], classroot.m.unit, classroot.m.id == "expenses") },
    open(p: number, s: number) {
        const per = s == 0 ? page?.periods[p] : page?.prevPeriods[p]
        if (per == null) return
        app.drill(per.from, per.to, m.kinds, null, null, null)
        },
    layout: SimpleLayout [ axis = y, spacing = 12 ],
    controls: View [ width = 100%,
        layout: WrappingLayout [ spacing = 10, rowSpacing = 8, align = center ],
        Segmented [ visible = { !app.phone }, height = { app.touchFirst ? 44 : 30 }, width = { Math.min(parent.contentBox("width"), 620) }, value = { app.trendMeasure }, accessibilityLabel = "Measure",
            items = { app.measures.map((x) => ({ id: x.id, label: x.name })) },
            commit(v: object) { app.trendMeasure = "" + v }
            ],
        DropButton [ visible = { app.phone }, label = { classroot.m.name }, accessibilityLabel = { "Measure: " + classroot.m.name },
            menu = { app.measureMenu } ],
        ByPicker [ ]
        ],
    frame: ChartFrame [ title = { classroot.m.name + " over time" }, sentence = { classroot.sentence }, source = { classroot.page?.cur ?? null },
        tableShown = { app.insightsTable },
        TableToggle [ ],
        Legend [ items = { classroot.series } ],
        TimeChart [ visible = { !app.insightsTable }, width = 100%, height = { app.phone ? 220 : 300 }, title = { classroot.m.name + " over time" },
            unit = { classroot.m.unit }, periods = { classroot.page?.chartPeriods ?? [] }, series = { classroot.series },
            opacity = { classroot.page?.cur.loading ? 0.6 : 1 },
            open(p: number, s: number) { classroot.open(p, s) },
            pickRange(a: number, b: number) { const ps = classroot.page?.chartPeriods ?? []; app.offerRange(ps[a].from, ps[b].to) }
            ],
        ChartTable [ visible = { app.insightsTable }, periods = { classroot.page?.chartPeriods ?? [] }, series = { classroot.series }, unit = { classroot.m.unit },
            open(p: number, s: number) { classroot.open(p, s) }
            ]
        ]
    ]


class IncomeView [ width = 100%,
    page: InsightsPage? = null,
    ps: Period[] = { classroot.page?.periods ?? [] },
