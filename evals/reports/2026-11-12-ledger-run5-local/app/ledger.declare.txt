/*
# Ledger

The books of a small business that makes and sells things: what came in, what
went out, and how the business is doing. Four screens over one service —
Overview, Transactions, Insights and Budgets — and one editor for recording
a sale, a refund or an expense.

How it is put together:

- **The service is the truth.** Every record and every total comes from the
  Ledger API (task/api/API.md); `Books` (ledger-model.declare) holds the
  lookups, `Writes` every change. Each accepted write bumps
  `writes.revision`, which every read's address carries, so every list,
  chart and total re-reads at once.
- **One set of filters**, on the App: the date range and its comparison,
  search, amount, kind, category, contact (and product, which a chart can
  set). `fq` is that filter as a query string, and every read is addressed
  by it, so every screen agrees and every total reconciles with the list.
  A newer request supersedes an older one, so an answer for filters since
  replaced is never shown; the previous answer stays on screen, dimmed,
  until the new one lands.
- **Where the person is**, on the App: the page (in the URL), the
  selection, what is open, the Insights tab. Opening a chart's figure sets
  the filters to exactly its records and goes to Transactions; `back`
  remembers where it came from so Back returns to the chart as it was.
- **The files**: ledger-model.declare (schemas, `Books`, `Writes`),
  ledger-screens.declare (the shell, the filter bar, Transactions and its
  cells and detail), ledger-reports.declare (Overview, Insights, Budgets),
  ledger-filters.declare (the filter panels), ledger-charts.declare,
  ledger-editor.declare (recording), ledger-parts.declare (themes and small
  parts), and ledger-format.ts (money, dates, ranges, CSV — pure functions).
*/

// The one thing a script does here that the language doesn't: hand the
// browser a file to save.

script [ host = browser ] {
    function saveFile(name: string, text: string, type: string) {
        const url = URL.createObjectURL(new Blob([text], { type: type }))
        const a = document.createElement("a")
        a.href = url
        a.download = name
        document.body.appendChild(a)
        a.click()
        a.remove()
        setTimeout(() => URL.revokeObjectURL(url), 1000)
    }
    }

    insightsTab: string = "trends",
    snapshot(label: string) -> FilterState {

        return both.length > 0 ? both : kinds

        location = "transactions"
        },

        if (s == null) return
        fPreset = s.preset; fFrom = s.from; fTo = s.to; fQ = s.q; fMin = s.min; fMax = s.max
        fKinds = s.kinds; fCats = s.cats; fContacts = s.contacts; fProducts = s.products
        insightsTab = s.tab
        back = null
        location = s.page
        },

    drillAll(cats: string[]) {
        back = this.snapshot(pageTitle)
        fPreset = "all"
        fKinds = []
        fCats = cats
        selection = []
        detailOpen = false
        location = "transactions"
        },

    openTrend(measure: string) {

    // ── the one plain sentence each chart opens with

    peakSentence(name: string, values: number[], periods: ChartPeriod[], unit: string, cost: boolean) -> string {
        const full = values.map((v, i) => ({ v: v, i: i })).filter((x) => !(periods[x.i]?.partial ?? false))
        const pool = full.length > 0 ? full : values.map((v, i) => ({ v: v, i: i }))
        if (pool.length < 2) return pool.length == 0 ? "Too little data in this range to draw a trend." : "Only one period in this range — widen the dates to see a trend."
        const best = pool.reduce((a, b) => b.v > a.v ? b : a)
        const fmt = unit == "percent" ? percent(best.v) : unit == "count" ? count(best.v) : money(best.v)
        const verb = name.endsWith("s") && unit != "count" ? " were " : " was "
        return name + verb + "highest in " + (periods[best.i]?.long ?? "") + " (" + fmt + ")" + (full.length < values.length ? "; the last period isn't over yet." : ".")
        },
    incomeSentence(ps: Period[]) -> string {
        if (ps.length == 0) return ""
        const net = ps.reduce((n, p) => n + p.net.amount, 0)
        const best = ps.reduce((a, b) => b.net.amount > a.net.amount ? b : a)
        return "Over these " + ps.length + " months, net profit was " + money(net) + "; the best month was " + periodLabel(best.period, "month", true) + " (" + money(best.net.amount) + ")."
        },
    breakEvenSentence(running: number[], periods: ChartPeriod[]) -> string {
        if (running.length == 0) return "Nothing in this range yet."
        const last = running[running.length - 1]
        let i = running.length - 1
        while (i > 0 && running[i - 1] >= 0) i = i - 1
        if (last < 0) return "Net profit is " + money(last) + " over the range: not yet broken even."
        if (running.every((v) => v >= 0)) return "In profit from " + (periods[0]?.long ?? "the start") + " on: " + money(last) + " net over the range."
        return "The running total of net profit passed zero in " + (periods[i]?.long ?? "") + " — the business broke even then, and stands at " + money(last) + "."
        },

    //
    // All of them, in the order asked for: a few thousand rows at most, so

    sortDir: string = "desc",
    txns: Read [ schema = TxnPage, path = { "/api/transactions?" + app.fq + "&sort=" + app.sortOn + "&dir=" + app.sortDir + "&limit=10000" } ],
    rows: Txn[] = { app.txns.value?.rows ?? [] },
    summary: Summary? = { app.txns.value?.summary ?? null },

    // ── the selection: records, held here so it survives scrolling and sorts.

    allMatching: boolean = { app.selection.length > 1 && app.selection.length == app.rows.length },

    detailOpen: boolean = false,

    trackChanges = ["rows", "fq"],
    onChange(e: ChangeEvent) {
        if (e.changed.some((c) => c.name == "fq")) { selection = []; return }
        const ids = new Set(selection.map((t) => t.id))
        selection = rows.filter((t) => ids.has(t.id))
        },
    selectAll() { selection = [...rows] },

    // a press inside a row's cell: the table's click protocol (⌘ toggles,
    // Shift extends), and a plain click opens the detail

    grid: TxnGrid? = { app.page == "transactions" ? app.shell.main.pages.transactions?.body.listCol.grid ?? null : null },
    rowPressed(t: Txn, e: PointerEvent) {
        const g = grid
        if (g == null) return
        const i = rows.findIndex((r) => r.id == t.id)
        g.rowClick(i, t, e.meta || e.ctrl, e.shift)
        g.activeIndex = i
        g.activeItem = t
        Focus.focus(g)                  // the arrows are the table's
        if (e.meta || e.ctrl || e.shift) return
        // Opening the detail narrows the table, so when it isn't open yet it
        // waits out a double-click — which edits the cell instead.
        if (detailOpen) return
        afterDelay(280, () => { if (Date.now() - app.editStartedAt > 400 && app.selection.length == 1) app.detailOpen = true })
        },
    editStartedAt: number = 0,

    // the summary line: what is shown, or the selection

    selSummary: Summary = { app.sumOf(app.selection) },
    sumOf(list: Txn[]) -> Summary {
        let a = 0
        let b = 0
        for (const t of list) { if (t.inFilter > 0) a = a + t.inFilter; else b = b - t.inFilter }
        return ({ count: list.length, in: a, out: b, net: a - b })
        },
    summaryLine(s: Summary?) -> string {
        if (s == null) return ""
        return count(s.count) + (s.count == 1 ? " transaction" : " transactions") + " · In " + money(s.in) + " · Out " + money(s.out) + " · Net " + money(s.net)
        },

    // ── bulk changes on the selection, each with Undo

    where() -> object {
        return allMatching ? filterQuery(this.spec(from, to)) : ({ ids: selection.map((t) => t.id) })
        },
    bulkCategorise(id: string?) {
        writes.runBulk(({ action: "categorise", categoryId: id, where: this.where(), expectedCount: selection.length }), "Categorised")
        },
    bulkContact(id: string?) {
        writes.runBulk(({ action: "set-contact", contactId: id, where: this.where(), expectedCount: selection.length }), "Set the contact on")
        },
    bulkDelete() {
        writes.runBulk(({ action: "delete", where: this.where(), expectedCount: selection.length }), "Deleted")
        selection = []
        detailOpen = false
        },

    // ── in-place edits
    //
    // An edit is sent at once; until the service answers, and if it
    // refuses, the cell keeps what was typed, marked as not saved, and the
    // bar under the list offers Retry.

    unsavedId: string = "",
    unsavedField: string = "",
    unsavedText: string = "",
    saveEdit(t: Txn, field: string, value: object, shown: string) {
        unsavedId = t.id
        unsavedField = field
        unsavedText = shown
        writes.update(t.id, ({ [field]: value }), "Saved")
        },
    unsaved(t: Txn, field: string) -> string {
        return unsavedId == t.id && unsavedField == field && (writes.patch.loading || writes.patch.failed) ? unsavedText : ""
        },
    editFailed: boolean = { app.unsavedId != "" && app.writes.patch.failed },
    retryEdit() { writes.patch.fetch() },
    dropEdit() { unsavedId = ""; writes.patch.clear() },

    // the category or contact chooser: one popover, opened at a cell (or for
    // the selection, from the bulk bar)

    pickWhat: "category" | "contact" = "category",
    pickTxn: Txn? = null,
    choose(what: "category" | "contact", t: Txn?, at: View) {
        pickWhat = what
        pickTxn = t
        pickPop.showFor(at)
        },
    chosen(id: string?) {
        const t = pickTxn
        pickPop.close()
        if (t == null) {
            if (pickWhat == "category") this.bulkCategorise(id); else this.bulkContact(id)
            return
            }
        if (pickWhat == "category") writes.update(t.id, ({ categoryId: id }), "Saved")
        else writes.update(t.id, ({ contactId: id }), "Saved")
        },

    // ── what the toast says, with Undo; ⌘Z undoes the last change too

    announceChange(words: string, undoable: boolean) {
        if (undoable) toast.say(words, ({ id: "undo", label: "Undo" }))
        else toast.say(words)
        },

        toast.say("Exported " + count(list.length) + (list.length == 1 ? " transaction" : " transactions"))
        },

        },
    refund(saleId: string) { refundFor = saleId; editorKind = "refund"; editorOpen = true },

    pickPop: Popover [ accessibilityLabel = { app.pickWhat == "category" ? "Choose a category" : "Choose a contact" },
        Chooser [ exists = { app.pickPop.present }, width = 280, what = { app.pickWhat }, txn = { app.pickTxn },
            picked(id: string?) { app.chosen(id) }
            ]
        ],

    // the editor: a panel at the side on a desk, a sheet from the bottom on a phone

    editor: Sheet [ edge = { app.phone ? "bottom" : "end" }, sheetWidth = 460, accessibilityLabel = "Record a transaction",
        shown = { app.editorOpen },
        commit(open: boolean) { app.editorOpen = open },
        Editor [ exists = { app.editor.present }, kind = { app.editorKind }, refundOf = { app.refundFor } ]
        ],

    toast: Toast [ duration = 6000,
        picked(id: string) {
            if (id == "undo") app.writes.undo()
            else if (id == "range" && app.offered != null) app.setDates(app.offered.from, app.offered.to)
            }
        ]
    ]
