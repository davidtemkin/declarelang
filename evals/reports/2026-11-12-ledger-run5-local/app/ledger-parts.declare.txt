/*
# Ledger: the parts

The look of Ledger and the small pieces its screens are built from: the two
themes, money as text, the category chip, filter chips, cards, and the
in-place states every panel shares (loading, failed with Retry, empty).
How the app works is in `ledger.declare`.
*/

// a search hit inside a description

style Hit [ fontWeight = semibold ]

// ── Money and numbers ───────────────────────────────────────────────────────
//
// Money is shown with every digit the same width, so columns line up.

class Money extends Text [ wrap = false, numeralWidth = tabular,
    cents: number = 0,
    text = { money(cents) }
    ]


// ── The category chip ───────────────────────────────────────────────────────
//
// A category in its colour; "Split (3)" for a split; "Uncategorised" marked
// as needing attention (an outline and a "!" as well as the colour).

class CatChip [ height = 22, cornerRadius = 11, padding = [0, 9, 0, 7],
    categoryId: string? = null,
    splitParts: number = 0,
    name: string = { classroot.splitParts > 0 ? "Split (" + classroot.splitParts + ")" : app.books.catName(classroot.categoryId) },
    colourIndex: number = { classroot.splitParts > 0 ? 10 : app.books.catColour(classroot.categoryId) },
    missing: boolean = { classroot.splitParts == 0 && classroot.categoryId == null },
    fill = { provided("theme").catTints[Math.max(0, classroot.colourIndex)] },
    State [ applied = { classroot.missing }, fill = { provided("theme").warnTint }, stroke = { stroke(1, provided("theme").warn) } ],
    tooltipLabel = { classroot.missing ? "Needs a category" : "" },
    layout: SimpleLayout [ axis = x, spacing = 5, align = center ],
    dot: View [ visible = { !classroot.missing }, width = 7, height = 7, cornerRadius = 3.5,
        fill = { provided("theme").cats[Math.max(0, classroot.colourIndex)] } ],
    bang: Text [ visible = { classroot.missing }, fontSize = 11, fontWeight = bold, wrap = false,
        textColor = { provided("theme").warn }, text = "!" ],
    Text [ fontSize = 12, wrap = false, maxLines = 1,
        textColor = { classroot.missing ? provided("theme").warn : provided("theme").text },
        text = { classroot.name } ]
    ]


// A coloured square for a legend or a list

class Swatch [ width = 10, height = 10, cornerRadius = 3,
    tone: Color = gray,
    fill = { classroot.tone }
    ]


// ── Filter chips ────────────────────────────────────────────────────────────
//
// An active filter: what it is and its value, and an × that removes it.

class FilterChip [ height = { app.touchFirst ? 36 : 28 }, cornerRadius = 14, padding = [0, 4, 0, 11],
    label: string = "",
    fill = { provided("theme").controlSelected },
    remove() { },
    layout: SimpleLayout [ axis = x, spacing = 2, align = center ],
    Text [ fontSize = 12.5, wrap = false, text = { classroot.label } ],
    close: Control [ width = { app.touchFirst ? 36 : 22 }, height = { this.width }, cornerRadius = { this.width / 2 },
        accessibilityLabel = { "Remove filter " + classroot.label },
        fill = { hovered || pressed ? provided("theme").controlHovered : null },
        press() { classroot.remove() },
        TextLabel [ x = center, fontSize = 13, textColor = { provided("theme").textMuted }, text = "×" ]
        ]
    ]


// A quiet button: text, a hover fill, no frame

class QuietButton extends Control [ height = { app.tap }, cornerRadius = 7, padding = [0, 10, 0, 10],
    label: string = "",
    tone: Color = { provided("theme").text },
    fill = { pressed ? provided("theme").controlPressed : hovered ? provided("theme").control : null },
    TextLabel [ y = center, fontSize = 13, wrap = false, textColor = { classroot.tone }, text = { classroot.label } ]
    ]


// A menu button: a label and a chevron; `on` marks a value that is set

class DropButton extends Control [ height = { app.tap }, cornerRadius = 7, padding = [0, 8, 0, 11],
    label: string = "",
    on: boolean = false,
    fill = { classroot.on ? provided("theme").controlSelected : hovered || pressed ? provided("theme").controlHovered : provided("theme").control },
    layout: SimpleLayout [ axis = x, spacing = 6, align = center ],
    TextLabel [ fontSize = 13, wrap = false, text = { classroot.label } ],
    ChevronIcon [ iconSize = 10, ink = { provided("theme").textMuted } ]
    ]


// ── Cards and panels ────────────────────────────────────────────────────────

class Panel [ width = 100%, cornerRadius = 12, padding = 16,
    fill = { provided("theme").surface }, stroke = { stroke(1, provided("theme").line) },
    title: string = "",
    defaultPlacement = body,
    layout: SimpleLayout [ axis = y, spacing = 10 ],
    head: View [ width = 100%, visible = { classroot.title != "" },
        layout: SimpleLayout [ axis = x, spacing = 8, align = center ],
        Text [ flexes = true, fontSize = 14, fontWeight = semibold, wrap = false, accessibilityRole = heading,
            text = { classroot.title } ]
        ],
    body: View [ width = 100%,
        layout: SimpleLayout [ axis = y, spacing = 8 ]
        ]
    ]


// The state of one load, said in place: a failure with Retry, or a slow
// answer still coming. The rest of the screen keeps working.

class LoadState [ width = 100%,
    source: Read? = null,
    slow: boolean = { classroot.source?.slow ?? false },
    failed: boolean = { classroot.source != null && classroot.source.failed },
    visible = { classroot.failed || classroot.slow },
    layout: SimpleLayout [ axis = x, spacing = 10, align = center ],
    Text [ flexes = true, fontSize = 13, wrap = true,
        textColor = { classroot.failed ? provided("theme").bad : provided("theme").textMuted },
        text = { classroot.failed ? "Couldn't load this. " + (classroot.source?.failure == "networkError" || classroot.source?.failure == "offline" ? "The service isn't answering." : "") : "Still working…" } ],
    Button [ visible = { classroot.failed }, label = "Retry", height = 28,
        onClick() { classroot.source?.fetch() }
        ]
    ]


// A small horizontal bar with a label and a figure: top products, top
// customers. Pressing it opens its records.

class BarItem extends Control [ width = 100%, height = 34, cornerRadius = 6, padding = [0, 6, 0, 6],
    label: string = "",
    value: number = 0,
    peak: number = 1,
    tone: Color = gray,
    figure: string = { money(classroot.value) },
    fill = { hovered || pressed ? provided("theme").rowHover : null },
    layout: SimpleLayout [ axis = y, spacing = 3 ],
    line: View [ width = 100%,
        layout: SimpleLayout [ axis = x, spacing = 8, align = baseline ],
        Text [ flexes = true, fontSize = 13, wrap = false, maxLines = 1, text = { classroot.label } ],
        Text [ fontSize = 13, wrap = false, numeralWidth = tabular, text = { classroot.figure } ]
        ],
    track: View [ width = 100%, height = 5, cornerRadius = 2.5, fill = { provided("theme").sunken },
        View [ height = 5, cornerRadius = 2.5, fill = { classroot.tone },
            width = { Math.max(2, parent.width * Math.max(0, classroot.value) / Math.max(1, classroot.peak)) } ]
        ]
    ]
