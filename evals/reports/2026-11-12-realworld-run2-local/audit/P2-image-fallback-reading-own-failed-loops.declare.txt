// An avatar that falls back to the default picture when its URL fails to load.
// a: reads its own `failed` (the obvious one-Image fallback); b: the default under, the picture over.
class Avatar extends Image [ width = 40, height = 40,
    url: string = "",
    source = { classroot.url == "" || classroot.failed ? "default-avatar.svg" : classroot.url }
    ]
class Layered [ width = 40, height = 40,
    url: string = "",
    Image [ width = 40, height = 40, source = "default-avatar.svg" ],
    pic: Image [ width = 40, height = 40, source = { classroot.url }, visible = { classroot.url != "" && !this.failed } ]
    ]
App [ width = 200, height = 100,
    a: Avatar [ url = "javascript:alert(1)" ]
    ]
